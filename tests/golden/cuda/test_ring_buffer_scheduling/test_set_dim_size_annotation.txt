// We need this inline namespace to avoid ODR problems in pytest.
inline namespace exo_CudaInline_test_set_dim_size_annotation {
struct exo_Cuda0_p
{
  using exo_DeviceArgs = exo_CudaDeviceArgs0_p;

  static constexpr uint32_t exo_blockDim = 256;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 256;
  static constexpr unsigned exo_smemOffset0_my_barrier = 0;  // 256-byte allocation

  struct exo_Task
  {
    int_fast32_t task;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_task;
    uint32_t exo_cudaTasksNum_task;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_task, int_fast32_t _exo_cudaTasksHi_task,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_task = _exo_cudaTasksLo_task;
      exo_cudaTasksNum_task = static_cast<uint32_t>(_exo_cudaTasksHi_task - _exo_cudaTasksLo_task);
      exo_taskCount *= exo_cudaTasksNum_task;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.task = exo_cudaTasksLo_task + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_task);
      exo_tmp /= exo_cudaTasksNum_task;
      return exo_task;
    }
  };

  struct exo_SyncState
  {
    int ring_consumption_0_my_barrier = 0;
    // my_barrier: barrier @ CudaMbarrier
    // num_per_cta=32; arrive_count=32
    EXO_CUDA_INLINE uint32_t Arrive0_my_barrier(char* exo_smem, exo_ExcutThreadLog exo_excutLog, int i0, int i1, int i2, bool enable) {
      const int mbarrier_idx = 8 * i0 + 4 * i1 + 1 * (int(unsigned(i2) % 4));
      const auto mbarrier_u32 = exo_smemU32(exo_smem + exo_smemOffset0_my_barrier + 8*mbarrier_idx);
      if (enable) {
        asm volatile(
          "// Arrive0_my_barrier\n\t"
          "mbarrier.arrive.shared::cta.b64 _, [%0];"
            :
            :"r"(mbarrier_u32)
        );
        exo_excutLog.log_action(EXO_EXCUT_STR_ID(mbarrier_arrive_shared_cta_b64), 0, __LINE__);
        exo_excutLog.log_u32_arg(static_cast<uint32_t>(mbarrier_u32));
      }
      return mbarrier_u32;
    }
    EXO_CUDA_INLINE void Await0_my_barrier(char* exo_smem, exo_ExcutThreadLog exo_excutLog, int i0, int i1, int i2) {
      const int mbarrier_idx = 8 * i0 + 4 * i1 + 1 * (int(unsigned(i2) % 4));
      const auto mbarrier_u32 = exo_smemU32(exo_smem + exo_smemOffset0_my_barrier + 8*mbarrier_idx);
      const unsigned parity = (unsigned(i2) / 4) & 1;
      const bool enable = true;
      if (enable) {
    #if __CUDA_ARCH__ < 900
        asm volatile(
          "{\n\t"
          "// Await0_my_barrier\n\t"
          ".reg.pred P1;\n\t"
          "EXO_BEFORE_WAIT:\n\t"
          "mbarrier.test_wait.parity.acquire.cta.shared::cta.b64 P1, [%0], %1;\n\t"
          "@P1 bra.uni EXO_WAIT_DONE;\n\t"
          "bra.uni EXO_BEFORE_WAIT;\n\t"
          "EXO_WAIT_DONE:\n\t"
          "}"
            :
            :"r"(mbarrier_u32),
            "r"(parity)
        );
        exo_excutLog.log_action(EXO_EXCUT_STR_ID(mbarrier_test_wait_parity_acquire_cta_shared_cta_b64), 0, __LINE__);
        exo_excutLog.log_u32_arg(static_cast<uint32_t>(mbarrier_u32));
        exo_excutLog.log_u32_arg(static_cast<uint32_t>(parity));
    #else
        asm volatile(
          "{\n\t"
          "// Await0_my_barrier\n\t"
          ".reg.pred P1;\n\t"
          "EXO_BEFORE_WAIT:\n\t"
          "mbarrier.try_wait.parity.acquire.cta.shared::cta.b64 P1, [%0], %1;\n\t"
          "@P1 bra.uni EXO_WAIT_DONE;\n\t"
          "bra.uni EXO_BEFORE_WAIT;\n\t"
          "EXO_WAIT_DONE:\n\t"
          "}"
            :
            :"r"(mbarrier_u32),
            "r"(parity)
        );
        exo_excutLog.log_action(EXO_EXCUT_STR_ID(mbarrier_try_wait_parity_acquire_cta_shared_cta_b64), 0, __LINE__);
        exo_excutLog.log_u32_arg(static_cast<uint32_t>(mbarrier_u32));
        exo_excutLog.log_u32_arg(static_cast<uint32_t>(parity));
    #endif
      }
    }
  };

  static inline const char*& exo_FILE()
  {
    static const char* name = __FILE__;
    return name;
  }

  static void
  exo_cudaLaunch(cudaStream_t exo_cudaStream, const exo_DeviceArgs& exo_deviceArgs);

  static __device__ __forceinline__ void
  exo_deviceSetup(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceShutdown(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceMainLoop(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceTask(
      char* exo_smem,
      exo_SyncState& exo_syncState,
      const exo_DeviceArgs& exo_deviceArgs,
      exo_Task exo_task,
      exo_ExcutThreadLog exo_excutLog={});
};
}  // end inline namespace

inline void
exo_CudaInline_test_set_dim_size_annotation::exo_Cuda0_p::exo_cudaLaunch(
    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_set_dim_size_annotation;
  cudaFuncSetAttribute(exo_deviceFunction0_p, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 1u;

  cudaLaunchConfig_t exo_launchConfig = {};
  exo_launchConfig.gridDim = dim3(exo_gridDim, 1, 1);
  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

  cudaLaunchKernelEx(&exo_launchConfig, exo_deviceFunction0_p, exo_deviceArgs);

  exo_excut_flush_device_log(
      exo_cudaStream, exo_gridDim, exo_blockDim,
      exo_CudaUtil::exo_excut_str_id_count, exo_CudaUtil::exo_excut_str_table,
      1, &exo_FILE());
}

__device__ __forceinline__ void
exo_CudaInline_test_set_dim_size_annotation::exo_Cuda0_p::exo_deviceSetup(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  if (threadIdx.x == 0) {
    for (int i = 0; i < 32; ++i) {
        asm volatile(
          "mbarrier.init.shared::cta.b64 [%0], 32;"
            :
            :"r"(exo_smemU32(exo_smem + exo_smemOffset0_my_barrier + 8*i))
        );
        exo_excutLog.log_action(EXO_EXCUT_STR_ID(mbarrier_init_shared_cta_b64), 0, __LINE__);
        exo_excutLog.log_u32_arg(exo_smemU32(exo_smem + exo_smemOffset0_my_barrier + 8*i));
        exo_excutLog.log_u32_arg(static_cast<uint32_t>(32));
    }
  }
  asm volatile(
    "barrier.cta.sync 0;"
  );
  exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
  exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
}

exo_CudaInline_test_set_dim_size_annotation::exo_Cuda0_p::exo_deviceShutdown(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  // clusterDim = 1
}

__device__ __forceinline__ void
exo_CudaInline_test_set_dim_size_annotation::exo_Cuda0_p::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_set_dim_size_annotation;
  // my_barrier: barrier[4, 2, 4] @ CudaMbarrier
  for (int k = 0; k < exo_deviceArgs.K; k++) {
    // cuda_threads(0, 4, unit=2 * cuda_warp)
    if ([[maybe_unused]] int exo_64thr_x = (threadIdx.x / 64); 1) {
      // cuda_threads(0, 2, unit=cuda_warp)
      if ([[maybe_unused]] int exo_32thr_y = (threadIdx.x % 64 / 32); 1) {
        float some_scalar;
        some_scalar = 1337.0f;
        // Arrive(cuda_in_order, 1) >> my_barrier[x, y, ((k + exo_syncState.ring_consumption_0_my_barrier) % 4)]
        // cta_mask: uint16_t(0x1)
        exo_syncState.Arrive0_my_barrier(exo_smem, exo_excutLog, int(exo_64thr_x), int(exo_32thr_y), int((k + (exo_syncState.ring_consumption_0_my_barrier))), 1);
        // Await(my_barrier[x, y, ((k + exo_syncState.ring_consumption_0_my_barrier) % 4)], cuda_in_order, 0)
        exo_syncState.Await0_my_barrier(exo_smem, exo_excutLog, int(exo_64thr_x), int(exo_32thr_y), int((k + (exo_syncState.ring_consumption_0_my_barrier))));
      }
    }
  }
  // free(my_barrier)
  exo_syncState.ring_consumption_0_my_barrier = __shfl_sync(UINT32_MAX,
      exo_syncState.ring_consumption_0_my_barrier + exo_deviceArgs.K - 0, 0);
}
__device__ __forceinline__ void
exo_CudaInline_test_set_dim_size_annotation::exo_Cuda0_p::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_set_dim_size_annotation;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 256) {
      {
        int_fast32_t exo_cudaTasksLo_task = 0;
        int_fast32_t exo_cudaTasksHi_task = 1;
        exo_TaskGenerator exo_taskGenerator(
            blockIdx.x / exo_clusterDim,
            gridDim.x / exo_clusterDim,
            exo_cudaTasksLo_task, exo_cudaTasksHi_task,
            exo_deviceArgs);
        while (exo_taskGenerator.prepare_next_task()) {
          exo_deviceTask(exo_smem, exo_syncState, exo_deviceArgs, exo_taskGenerator.get_next_task(), exo_excutLog);
        }
      }
    }
  }
}

__launch_bounds__(256, 1)
__global__ void
exo_deviceFunction0_p(__grid_constant__ const struct exo_CudaDeviceArgs0_p exo_deviceArgs)
{
  extern __shared__ __align__(128) char exo_smem[];
  exo_ExcutThreadLog exo_excutLog = exo_excut_begin_thread_log(exo_deviceArgs.exo_excutDeviceLog);
  exo_Cuda0_p::exo_deviceSetup(exo_smem, exo_deviceArgs, exo_excutLog);
  exo_Cuda0_p::exo_deviceMainLoop(exo_smem, exo_deviceArgs, exo_excutLog);
  exo_Cuda0_p::exo_deviceShutdown(exo_smem, exo_deviceArgs, exo_excutLog);
}

exo_cudaLaunch0_p(cudaStream_t exo_cudaStream, const struct exo_CudaDeviceArgs0_p* exo_deviceArgs)
{
  exo_Cuda0_p::exo_cudaLaunch(exo_cudaStream, *exo_deviceArgs);
}
