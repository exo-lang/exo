// _6_256_8_cp_async_commit_group(
//     M : size,
//     K : size,
//     h_out : f32[M, K] @DRAM,
//     h_in : f32[M, K] @DRAM
// )
void _6_256_8_cp_async_commit_group( void *ctxt, int_fast32_t M, int_fast32_t K, float* h_out, const float* h_in ) {
  EXO_ASSUME(M % 256 == 0);
  EXO_ASSUME(K % 8 == 0);
  float *d_in = (float*) exo_cudaMallocAsync(M * K * sizeof(*d_in), exo_cudaStream);
  float *d_out = (float*) exo_cudaMallocAsync(M * K * sizeof(*d_out), exo_cudaStream);
  cudaMemcpyAsync(
    (&d_in[0]),
    (&h_in[0]),
    M * K * 4,
    cudaMemcpyHostToDevice,
    exo_cudaStream);
  {
    struct exo_CudaDeviceArgs0__6_256_8_cp_async_commit_group exo_deviceArgs = {
      M, K, d_in, d_out, exo_excut_get_device_log()
    };
    exo_cudaLaunch0__6_256_8_cp_async_commit_group(exo_cudaStream, &exo_deviceArgs);
  }
  exo_cudaFreeAsync(d_in, exo_cudaStream);
  cudaMemcpyAsync(
    (&h_out[0]),
    (&d_out[0]),
    M * K * 4,
    cudaMemcpyDeviceToHost,
    exo_cudaStream);
  exo_cudaFreeAsync(d_out, exo_cudaStream);
}

exo_CudaInline_test_golden_B_cp_async_commit_group::exo_Cuda0__6_256_8_cp_async_commit_group::exo_deviceTask(

    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_golden_B_cp_async_commit_group;
  // cg: barrier @ Sm80_CommitGroup
  auto& smem = reinterpret_cast<float (&) [12288]>(exo_smem[exo_smemOffset0_smem]);
  for (int warmup = 0; warmup < 4; warmup++) {
    // cuda_threads(0, 256, unit=cuda_thread)
    if ([[maybe_unused]] int exo_1thr_tid = threadIdx.x; 1) {
      if (warmup < ((exo_deviceArgs.K) / (8))) {
        for (int k_cp = 0; k_cp < 2; k_cp++) {
          asm volatile(
            "cp.async.cg.shared.global [%0], [%1], 16;"
              :
              :"r"(exo_smemU32((&smem[warmup * 2048 + exo_1thr_tid * 8 + (4 * k_cp)]))),
              "l"((&exo_deviceArgs.d_in[(exo_1thr_tid + 256 * exo_task.m_task) * exo_deviceArgs.K + (4 * k_cp + 8 * warmup)]))
          );
          exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_cg_shared_global), 0, __LINE__);
          exo_excutLog.log_u32_arg(exo_smemU32((&smem[warmup * 2048 + exo_1thr_tid * 8 + (4 * k_cp)])));
          exo_excutLog.log_ptr_arg((&exo_deviceArgs.d_in[(exo_1thr_tid + 256 * exo_task.m_task) * exo_deviceArgs.K + (4 * k_cp + 8 * warmup)]));
          exo_excutLog.log_u32_arg(static_cast<uint32_t>(16));
        }
      }
      // Arrive(Sm80_cp_async, 1) >> cg
      asm volatile(
        "cp.async.commit_group;"
      );
      exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_commit_group), 0, __LINE__);
    }
  }
  for (int k_iter = 0; k_iter < ((exo_deviceArgs.K) / (8)); k_iter++) {
    // cuda_threads(0, 256, unit=cuda_thread)
    if ([[maybe_unused]] int exo_1thr_tid = threadIdx.x; 1) {
      if (4 + k_iter < ((exo_deviceArgs.K) / (8))) {
        for (int k_cp = 0; k_cp < 2; k_cp++) {
          asm volatile(
            "cp.async.cg.shared.global [%0], [%1], 16;"
              :
              :"r"(exo_smemU32((&smem[((4 + k_iter) % 6) * 2048 + exo_1thr_tid * 8 + (4 * k_cp)]))),
              "l"((&exo_deviceArgs.d_in[(exo_1thr_tid + 256 * exo_task.m_task) * exo_deviceArgs.K + (32 + 4 * k_cp + 8 * k_iter)]))
          );
          exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_cg_shared_global), 0, __LINE__);
          exo_excutLog.log_u32_arg(exo_smemU32((&smem[((4 + k_iter) % 6) * 2048 + exo_1thr_tid * 8 + (4 * k_cp)])));
          exo_excutLog.log_ptr_arg((&exo_deviceArgs.d_in[(exo_1thr_tid + 256 * exo_task.m_task) * exo_deviceArgs.K + (32 + 4 * k_cp + 8 * k_iter)]));
          exo_excutLog.log_u32_arg(static_cast<uint32_t>(16));
        }
      }
      // Arrive(Sm80_cp_async, 1) >> cg
      asm volatile(
        "cp.async.commit_group;"
      );
      exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_commit_group), 0, __LINE__);
      // Await(cg, cuda_in_order, 4)
      asm volatile(
        "cp.async.wait_group 4;"
      );
      exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_wait_group), 0, __LINE__);
      exo_excutLog.log_u32_arg(static_cast<uint32_t>(4));
    }
    // Fence(cuda_in_order, cuda_in_order)
    asm volatile(
      "barrier.cta.sync 0;"
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
    exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
    for (int ms = 0; ms < 8; ms++) {
      // cuda_threads(0, 32, unit=8 * cuda_thread)
