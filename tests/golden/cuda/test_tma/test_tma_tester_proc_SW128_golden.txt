template <typename T>
struct exo_Sm90_SW128 {
    T data;

    static EXO_CUDA_INLINE exo_Sm90_SW128<T>* swizzle_pointer(uintptr_t addr)
    {
        // Adapted from ThunderKittens appendix which actually documents CUDA correctly.
        uint32_t shr = uint32_t(addr) >> 3;
        const uint32_t mask = 112;
        addr = addr ^ (shr & mask);
        return reinterpret_cast<exo_Sm90_SW128*>(addr);
    }

    static __host__ __device__ constexpr uint64_t get_swizzle_bits()
    {
        return 1;
    }

    static __host__ __device__ constexpr int get_swizzle_bytes()
    {
        return 128;
    }

    EXO_CUDA_INLINE const T& swizzle_get() const
    {
        return swizzle_pointer(reinterpret_cast<uintptr_t>(&data))->data;
    }

    EXO_CUDA_INLINE T& swizzle_get()
    {
        return swizzle_pointer(reinterpret_cast<uintptr_t>(&data))->data;
    }
};

// Element type for Sm90_SmemSwizzled(128) allocations where the
// last 3 array extents are [TileOuterCols, TileRows, TileInnerCols]
// (left-padded with 1 for 0D, 1D, 2D allocations).
template <typename T, int TileOuterCols, int TileRows, int TileInnerCols>
struct exo_Sm90_SW128_tiled: public exo_Sm90_SW128<T>
{
    // Reinterpret-cast to kittens::st_subtile view of this shared memory.
    //
    // For better or worse, kittens 2D tile is expressed as a 3D tile in Exo-GPU.
    // `this` is assumed to point-to the NON-SWIZZLED base address of a complete
    // 3D tile. This returns a subtile view whose
    //
    // * size/extents are given by the Subtile template parameters
    //
    // * base offset from the `this` tile is given by runtime int offsets
    //
    // Assume for this discussion that the tile is row-major. Then,
    //
    // * Let swizzle_elements = 128 / sizeof(T)
    //
    // * The 3D Exo-GPU tile is of size [TileOuterCols, TileRows, TileInnerCols],
    //   where TileInnerCols == swizzle_elements. If not, then this tile
    //   is incompatible with ThunderKittens, TMA, and wgmma.
    //
    // * The value at coordinates (r, c) in the 2D [Rows, Cols] tile is stored at
    //   address swizzle_pointer(
    //          tile_base_addr
    //          + sizeof(T) * (c / TileInnerCols) * TileRows * TileInnerCols
    //          + sizeof(T) * (r) * TileInnerCols
    //          + sizeof(T) * (c % TileInnerCols))
    //   Basically, the column is simultaneously the fastest (%) and slowest (/) dimension
    //
    // TMA: If TileOuterCols != 1, then ThunderKittens is capable of doing a single TMA
    // copy to load/store the tile, but Exo-GPU may have to issue multiple.
    template <
        int SubtileOuterCols,
        int SubtileRows,
        int SubtileInnerCols,
        template <int, int, bool, int> class st_typed>
    EXO_CUDA_INLINE auto as_tk_subtile(int col_outer_offset, int row_offset, int col_inner_offset) const
    {
        static_assert(
            TileInnerCols * sizeof(T) == 128,
            "Exo-GPU instr didn't assert strides properly."
            " This is needed to match kittens swizzle automation"
        );
        static_assert(
            SubtileInnerCols == TileInnerCols || SubtileOuterCols == 1,
            "Exo-GPU instr used wrong SubtileInnerCols (window size last dimension)."
            " This is needed for Exo-GPU and kittens to agree on column tiling"
        );
        using st_t = st_typed<TileRows, TileOuterCols * TileInnerCols, true, 0>;
        st_t* p_tile = const_cast<st_t*>(reinterpret_cast<const st_t*>(this));
        // Note, subtile implicitly multiplies (r, c) by the subtile size.
        // We have to work around this!
        // Also, can't mention ::kittens here, because it may not be included.
        // Also also, explicit kittens swizzle is broken in this function.
        auto subtile = p_tile->template subtile<SubtileRows, SubtileOuterCols * SubtileInnerCols>(int2(0, 0));
        subtile.row_offset = static_cast<int>(row_offset);
        subtile.col_offset = static_cast<int>(col_outer_offset) * TileInnerCols + static_cast<int>(col_inner_offset);
        return subtile;
    }
};

/* Required by Sm90_tensorMap(128, 248, 32) */
static inline CUtensorMap exo_win_2f32_Sm90_tensorMap_128_248_32_encode(
        // Window dataptr, strides
        const void* globalAddress, exo_Sm90_CUtensorMap_2_strides gmem_stride,
        // Tensor size
        exo_Sm90_CUtensorMap_2_dim gmem_dim)
{
    assert(gmem_stride.C_strides[2 - 1] == 1);

    CUtensorMap tensorMap;
    const CUtensorMapSwizzle swizzle = CU_TENSOR_MAP_SWIZZLE_128B;

    cuuint64_t globalDim[2];
    cuuint64_t allGlobalStrides[2];  // allGlobalStrides[0] unused by CUDA
    cuuint32_t elementStrides[2];

    // We translate from the Exo ordering (leftmost stride is most-significant)
    // to the CUDA ordering (leftmost stride is least-significant).
    for (uint32_t cu_idx = 0; cu_idx < 2; ++cu_idx) {
        const uint32_t C_idx = 2 - 1 - cu_idx;
        globalDim[cu_idx] = gmem_dim.C_dim[C_idx];
        allGlobalStrides[cu_idx] = ((cuuint64_t)gmem_stride.C_strides[C_idx]) * 4;
        elementStrides[cu_idx] = 1;
    }

    cuuint32_t boxDim[2] = { 32, 248 };
    const CUtensorMapInterleave interleave = CU_TENSOR_MAP_INTERLEAVE_NONE;
    const CUtensorMapL2promotion l2Promotion = CU_TENSOR_MAP_L2_PROMOTION_L2_128B;
    const CUtensorMapFloatOOBfill oobFill = CU_TENSOR_MAP_FLOAT_OOB_FILL_NONE;

    const CUresult result = cuTensorMapEncodeTiled(
            &tensorMap,
            CU_TENSOR_MAP_DATA_TYPE_FLOAT32,
            2,
            (void*)globalAddress,
            globalDim,
            &allGlobalStrides[1],  // Cuda presumes least-significant dim is tightly-packed
            boxDim,
            elementStrides,
            interleave,
            swizzle,
            l2Promotion,
            oobFill);
    if (result != 0) {
        fprintf(stderr, "exo_win_2f32_Sm90_tensorMap_128_248_32_encode: error %i\n", (int)result);
        assert(0);
    }
    return tensorMap;
}

// tma_tester_SW128(
//     h_sum : f32[744, 128] @DRAM,
//     h_x : f32[744, 128] @DRAM,
//     h_y : f32[744, 128] @DRAM
// )
void tma_tester_SW128( void *ctxt, float* h_sum, const float* h_x, const float* h_y ) {
  float *d_x = (float*) exo_cudaMallocAsync(744 * 192 * sizeof(*d_x), exo_cudaStream);
  float *d_y = (float*) exo_cudaMallocAsync(744 * 128 * sizeof(*d_y), exo_cudaStream);
  float *d_sum = (float*) exo_cudaMallocAsync(745 * 128 * sizeof(*d_sum), exo_cudaStream);
  for (int_fast32_t m = 0; m < 744; m++) {
    cudaMemcpyAsync(
      (&d_x[m * 192]),
      (&h_x[m * 128]),
      128 * 4,
      cudaMemcpyHostToDevice,
      exo_cudaStream);
  }
  cudaMemcpyAsync(
    (&d_y[0]),
    (&h_y[0]),
    744 * 128 * 4,
    cudaMemcpyHostToDevice,
    exo_cudaStream);
  CUtensorMap exo_data_sum_tensorMap = exo_win_2f32_Sm90_tensorMap_128_248_32_encode(&d_sum[0], (exo_Sm90_CUtensorMap_2_strides){ {128, 1} }, (exo_Sm90_CUtensorMap_2_dim){ {745, 128} });
  struct exo_win_2f32_Sm90_tensorMap_128_248_32 sum_tensorMap = (exo_win_2f32_Sm90_tensorMap_128_248_32) { {0, 0} };
  CUtensorMap exo_data_sum_tensorMap_window = exo_data_sum_tensorMap;
  struct exo_win_2f32_Sm90_tensorMap_128_248_32 sum_tensorMap_window = (exo_win_2f32_Sm90_tensorMap_128_248_32) { {(sum_tensorMap.C_offsets[0] + 1), sum_tensorMap.C_offsets[1]} };
  tma_tester_gpu_SW128(ctxt, exo_data_sum_tensorMap_window, (exo_win_2f32_Sm90_tensorMap_128_248_32) { {sum_tensorMap_window.C_offsets[0], sum_tensorMap_window.C_offsets[1]} }, (struct exo_win_2f32c) { &d_x[0], {192, 1} }, d_y);
  exo_cudaFreeAsync(d_y, exo_cudaStream);
  exo_cudaFreeAsync(d_x, exo_cudaStream);
  cudaMemcpyAsync(
    (&h_sum[0]),
    (&d_sum[128]),
    744 * 128 * 4,
    cudaMemcpyDeviceToHost,
    exo_cudaStream);
  exo_cudaFreeAsync(d_sum, exo_cudaStream);
}

template <typename T>
struct exo_Sm90_SW128 {
    T data;

    static EXO_CUDA_INLINE exo_Sm90_SW128<T>* swizzle_pointer(uintptr_t addr)
    {
        // Adapted from ThunderKittens appendix which actually documents CUDA correctly.
        uint32_t shr = uint32_t(addr) >> 3;
        const uint32_t mask = 112;
        addr = addr ^ (shr & mask);
        return reinterpret_cast<exo_Sm90_SW128*>(addr);
    }

    static __host__ __device__ constexpr uint64_t get_swizzle_bits()
    {
        return 1;
    }

    static __host__ __device__ constexpr int get_swizzle_bytes()
    {
        return 128;
    }

    EXO_CUDA_INLINE const T& swizzle_get() const
    {
        return swizzle_pointer(reinterpret_cast<uintptr_t>(&data))->data;
    }

    EXO_CUDA_INLINE T& swizzle_get()
    {
        return swizzle_pointer(reinterpret_cast<uintptr_t>(&data))->data;
    }
};

// Element type for Sm90_SmemSwizzled(128) allocations where the
// last 3 array extents are [TileOuterCols, TileRows, TileInnerCols]
// (left-padded with 1 for 0D, 1D, 2D allocations).
template <typename T, int TileOuterCols, int TileRows, int TileInnerCols>
struct exo_Sm90_SW128_tiled: public exo_Sm90_SW128<T>
{
    // Reinterpret-cast to kittens::st_subtile view of this shared memory.
    //
    // For better or worse, kittens 2D tile is expressed as a 3D tile in Exo-GPU.
    // `this` is assumed to point-to the NON-SWIZZLED base address of a complete
    // 3D tile. This returns a subtile view whose
    //
    // * size/extents are given by the Subtile template parameters
    //
    // * base offset from the `this` tile is given by runtime int offsets
    //
    // Assume for this discussion that the tile is row-major. Then,
    //
    // * Let swizzle_elements = 128 / sizeof(T)
    //
    // * The 3D Exo-GPU tile is of size [TileOuterCols, TileRows, TileInnerCols],
    //   where TileInnerCols == swizzle_elements. If not, then this tile
    //   is incompatible with ThunderKittens, TMA, and wgmma.
    //
    // * The value at coordinates (r, c) in the 2D [Rows, Cols] tile is stored at
    //   address swizzle_pointer(
    //          tile_base_addr
    //          + sizeof(T) * (c / TileInnerCols) * TileRows * TileInnerCols
    //          + sizeof(T) * (r) * TileInnerCols
    //          + sizeof(T) * (c % TileInnerCols))
    //   Basically, the column is simultaneously the fastest (%) and slowest (/) dimension
    //
    // TMA: If TileOuterCols != 1, then ThunderKittens is capable of doing a single TMA
    // copy to load/store the tile, but Exo-GPU may have to issue multiple.
    template <
        int SubtileOuterCols,
        int SubtileRows,
        int SubtileInnerCols,
        template <int, int, bool, int> class st_typed>
    EXO_CUDA_INLINE auto as_tk_subtile(int col_outer_offset, int row_offset, int col_inner_offset) const
    {
        static_assert(
            TileInnerCols * sizeof(T) == 128,
            "Exo-GPU instr didn't assert strides properly."
            " This is needed to match kittens swizzle automation"
        );
        static_assert(
            SubtileInnerCols == TileInnerCols || SubtileOuterCols == 1,
            "Exo-GPU instr used wrong SubtileInnerCols (window size last dimension)."
            " This is needed for Exo-GPU and kittens to agree on column tiling"
        );
        using st_t = st_typed<TileRows, TileOuterCols * TileInnerCols, true, 0>;
        st_t* p_tile = const_cast<st_t*>(reinterpret_cast<const st_t*>(this));
        // Note, subtile implicitly multiplies (r, c) by the subtile size.
        // We have to work around this!
        // Also, can't mention ::kittens here, because it may not be included.
        // Also also, explicit kittens swizzle is broken in this function.
        auto subtile = p_tile->template subtile<SubtileRows, SubtileOuterCols * SubtileInnerCols>(int2(0, 0));
        subtile.row_offset = static_cast<int>(row_offset);
        subtile.col_offset = static_cast<int>(col_outer_offset) * TileInnerCols + static_cast<int>(col_inner_offset);
        return subtile;
    }
};

/* Required by Sm90_tma_load_2d(dst,src,dst=f32, src=f32, size0=248, size1=32, smem_box=None, swizzle=128) */
template <typename WindowOffsets>
EXO_CUDA_INLINE void
exo_Sm90_tma_to_smem(
        void* dst, const CUtensorMap& tensorMap, WindowOffsets window,
        uint32_t exo_tma_mbarrier, uint32_t expect_tx)
{
    constexpr auto rank = sizeof(window.C_offsets) / sizeof(window.C_offsets[0]);
    static_assert(rank >= 1 && rank <= 5);
    // cute::elect_one_sync
    uint32_t pred = 0;
    uint32_t laneid = 0;
    asm volatile(
      "{\n"
      ".reg .b32 %%rx;\n"
      ".reg .pred %%px;\n"
      "     elect.sync %%rx|%%px, %2;\n"
      "@%%px mov.s32 %1, 1;\n"
      "     mov.s32 %0, %%rx;\n"
      "}\n"
      : "+r"(laneid), "+r"(pred)
      : "r"(0xFFFFFFFF));
    if (pred) {
        asm("mbarrier.expect_tx.shared::cta.b64 [%0], %1;" :: "r"(exo_tma_mbarrier), "r"(expect_tx));
        if constexpr (rank == 1) {
            asm volatile(
            "cp.async.bulk.tensor.1d.shared::cluster.global.tile.mbarrier::complete_tx::bytes.L2::cache_hint"
            " [%0], [%1, {%2}], [%3], %4;"
            :
            : "r"(exo_smemU32(dst)), "l"(&tensorMap), "r"(window.C_offsets[0]),
              "r"(exo_tma_mbarrier), "n"(1152921504606846976)
            : "memory");
        }
        if constexpr (rank == 2) {
            asm volatile(
            "cp.async.bulk.tensor.2d.shared::cluster.global.tile.mbarrier::complete_tx::bytes.L2::cache_hint"
            " [%0], [%1, {%2, %3}], [%4], %5;"
            :
            : "r"(exo_smemU32(dst)), "l"(&tensorMap), "r"(window.C_offsets[1]), "r"(window.C_offsets[0]),
              "r"(exo_tma_mbarrier), "n"(1152921504606846976)
            : "memory");
        }
        if constexpr (rank == 3) {
            asm volatile(
            "cp.async.bulk.tensor.3d.shared::cluster.global.tile.mbarrier::complete_tx::bytes.L2::cache_hint"
            " [%0], [%1, {%2, %3, %4}], [%5], %6;"
            :
            : "r"(exo_smemU32(dst)), "l"(&tensorMap), "r"(window.C_offsets[2]), "r"(window.C_offsets[1]), "r"(window.C_offsets[0]),
              "r"(exo_tma_mbarrier), "n"(1152921504606846976)
            : "memory");
        }
        if constexpr (rank == 4) {
            asm volatile(
            "cp.async.bulk.tensor.4d.shared::cluster.global.tile.mbarrier::complete_tx::bytes.L2::cache_hint"
            " [%0], [%1, {%2, %3, %4, %5}], [%6], %7;"
            :
            : "r"(exo_smemU32(dst)), "l"(&tensorMap), "r"(window.C_offsets[3]), "r"(window.C_offsets[2]), "r"(window.C_offsets[1]), "r"(window.C_offsets[0]),
              "r"(exo_tma_mbarrier), "n"(1152921504606846976)
            : "memory");
        }
        if constexpr (rank == 5) {
            asm volatile(
            "cp.async.bulk.tensor.5d.shared::cluster.global.tile.mbarrier::complete_tx::bytes.L2::cache_hint"
            " [%0], [%1, {%2, %3, %4, %5, %6}], [%7], %8;"
            :
            : "r"(exo_smemU32(dst)), "l"(&tensorMap), "r"(window.C_offsets[4]), "r"(window.C_offsets[3]), "r"(window.C_offsets[2]), "r"(window.C_offsets[1]), "r"(window.C_offsets[0]),
              "r"(exo_tma_mbarrier), "n"(1152921504606846976)
            : "memory");
        }
    }
}

__device__ __forceinline__ void
exo_CudaInline_test_tma_tester_proc_SW128_golden::exo_Cuda0_tma_tester_gpu_SW128::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_tma_tester_proc_SW128_golden;
  auto& smem_x = reinterpret_cast<exo_Sm90_SW128_tiled<float, 1, 248, 32> (&)[]>(exo_smem[exo_smemOffset0_smem_x]);
  auto& smem_y = reinterpret_cast<exo_Sm90_SW128_tiled<float, 1, 248, 32> (&)[]>(exo_smem[exo_smemOffset1_smem_y]);
  auto& smem_sum = reinterpret_cast<exo_Sm90_SW128_tiled<float, 1, 248, 32> (&)[]>(exo_smem[exo_smemOffset2_smem_sum]);
  // raw: barrier[2] @ CudaMbarrier
  // war: barrier[2] @ CudaMbarrierPreArrive(1, )
  // CudaWarps(0, 1)
  if (int CudaWarps_0_1_ = threadIdx.x; CudaWarps_0_1_ < 32) {
    // Await(war[((0 + exo_syncState.ring_consumption_1_war) % 2)], cuda_temporal, 0)
    exo_syncState.Await0_war(exo_smem, exo_excutLog, int(((exo_syncState.ring_consumption_1_war) - 1)));
    CUtensorMap const& exo_data_x_input = exo_deviceArgs.exo_data_x_tensorMap_debug;
    struct exo_win_2f32_Sm90_tensorMap_128_248_32 x_input = (exo_win_2f32_Sm90_tensorMap_128_248_32) { {(exo_deviceArgs.x_tensorMap_debug.C_offsets[0] + 248 * exo_task.task_m), (exo_deviceArgs.x_tensorMap_debug.C_offsets[1] + 32 * exo_task.task_k)} };
    exo_CudaUtil::exo_Sm90_tma_to_smem(
        (&smem_x[0])
      , exo_data_x_input
      , (exo_win_2f32_Sm90_tensorMap_128_248_32) { {x_input.C_offsets[0], x_input.C_offsets[1]} }
      , exo_syncState.Arrive0_raw(exo_smem, exo_excutLog, int((exo_syncState.ring_consumption_0_raw)), 0)
      , 31744
    );
    // Arrive(cuda_temporal, 1) >> raw[((0 + exo_syncState.ring_consumption_0_raw) % 2)]
    // cta_mask: uint16_t(0x1)
    exo_syncState.Arrive0_raw(exo_smem, exo_excutLog, int((exo_syncState.ring_consumption_0_raw)), 1);
  }
  // cuda_threads(0, 248, unit=cuda_thread)
  if (int exo_1thr_m = threadIdx.x; exo_1thr_m < 248) {
    for (int ko = 0; ko < 8; ko++) {
      asm volatile(
        "cp.async.cg.shared.global [%0], [%1], 16;"
          :
          :"r"(exo_smemU32((&smem_y[(exo_1thr_m * 32 + 4 * ko)].swizzle_get()))),
          "l"((&exo_deviceArgs.d_y[(exo_1thr_m + 248 * exo_task.task_m) * 128 + (4 * ko + 32 * exo_task.task_k)]))
      );
      exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_cg_shared_global), 0, __LINE__);
      exo_excutLog.log_u32_arg(exo_smemU32((&smem_y[(exo_1thr_m * 32 + 4 * ko)].swizzle_get())));
      exo_excutLog.log_ptr_arg((&exo_deviceArgs.d_y[(exo_1thr_m + 248 * exo_task.task_m) * 128 + (4 * ko + 32 * exo_task.task_k)]));
      exo_excutLog.log_u32_arg(static_cast<uint32_t>(16));
    }
  }
  // Fence(Sm80_cp_async, cuda_in_order)
  asm volatile(
    "cp.async.wait_all;"
  );
  exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_wait_all), 0, __LINE__);
  asm volatile(
    "barrier.cta.sync 0;"
  );
  exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
  exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
  // Await(raw[((0 + exo_syncState.ring_consumption_0_raw) % 2)], cuda_in_order, 0)
  exo_syncState.Await0_raw(exo_smem, exo_excutLog, int((exo_syncState.ring_consumption_0_raw)));
  // cuda_threads(0, 248, unit=cuda_thread)
  if (int exo_1thr_m = threadIdx.x; exo_1thr_m < 248) {
    for (int k = 0; k < 32; k++) {
      smem_sum[(exo_1thr_m * 32 + k)].swizzle_get() = smem_x[(exo_1thr_m * 32 + k)].swizzle_get() + smem_y[(exo_1thr_m * 32 + k)].swizzle_get();
    }
  }
  // Arrive(cuda_in_order, 1) >> war[((1 + exo_syncState.ring_consumption_1_war) % 2)]
  // cta_mask: uint16_t(0x1)
  exo_syncState.Arrive0_war(exo_smem, exo_excutLog, int((1 + (exo_syncState.ring_consumption_1_war) - 1)), 1);
  // Fence(cuda_in_order, cuda_generic_and_async_proxy)
  asm volatile(
    "barrier.cta.sync 0;"
  );
  exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
  exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
  asm volatile(
    "fence.proxy.async;"
  );
  exo_excutLog.log_action(EXO_EXCUT_STR_ID(fence_proxy_async), 0, __LINE__);
  // CudaWarps(0, 1)
  if (int CudaWarps_0_1_ = threadIdx.x; CudaWarps_0_1_ < 32) {
    CUtensorMap const& exo_data_tma_window = exo_deviceArgs.exo_data_sum_tensorMap_window;
    struct exo_win_2f32_Sm90_tensorMap_128_248_32 tma_window = (exo_win_2f32_Sm90_tensorMap_128_248_32) { {(exo_deviceArgs.sum_tensorMap_window.C_offsets[0] + 248 * exo_task.task_m), (exo_deviceArgs.sum_tensorMap_window.C_offsets[1] + 32 * exo_task.task_k)} };
    exo_CudaUtil::exo_Sm90_tma_to_gmem(
        exo_data_tma_window
      , (exo_win_2f32_Sm90_tensorMap_128_248_32) { {tma_window.C_offsets[0], tma_window.C_offsets[1]} }
      , (&smem_sum[0])
    );
    // cg: barrier @ Sm90_TmaCommitGroup
    // Arrive(tma_to_gmem_async, 1) >> cg
    asm volatile(
      "cp.async.bulk.commit_group;"
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_bulk_commit_group), 0, __LINE__);
    // Await(cg, cuda_in_order, 0)
    asm volatile(
      "cp.async.bulk.wait_group 0;"
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(cp_async_bulk_wait_group), 0, __LINE__);
    exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
    // free(cg)
  }
  // Fence(cuda_in_order, cuda_in_order)
  asm volatile(
    "barrier.cta.sync 0;"
  );
  exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
  exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
  // free(war)
  exo_syncState.ring_consumption_1_war = __shfl_sync(UINT32_MAX,
      exo_syncState.ring_consumption_1_war + 2 - 1, 0);
  // free(raw)
  exo_syncState.ring_consumption_0_raw = __shfl_sync(UINT32_MAX,
      exo_syncState.ring_consumption_0_raw + 1 - 0, 0);
}
__device__ __forceinline__ void
exo_CudaInline_test_tma_tester_proc_SW128_golden::exo_Cuda0_tma_tester_gpu_SW128::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_tma_tester_proc_SW128_golden;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 256) {
      {
        int_fast32_t exo_cudaTasksLo_task_m = 0;
        int_fast32_t exo_cudaTasksHi_task_m = 3;
        {
          int_fast32_t exo_cudaTasksLo_task_k = 0;
          int_fast32_t exo_cudaTasksHi_task_k = 4;
          exo_TaskGenerator exo_taskGenerator(
              blockIdx.x / exo_clusterDim,
              gridDim.x / exo_clusterDim,
              exo_cudaTasksLo_task_m, exo_cudaTasksHi_task_m,
              exo_cudaTasksLo_task_k, exo_cudaTasksHi_task_k,
              exo_deviceArgs);
          while (exo_taskGenerator.prepare_next_task()) {
            exo_deviceTask(exo_smem, exo_syncState, exo_deviceArgs, exo_taskGenerator.get_next_task(), exo_excutLog);
          }
        }
      }
    }
  }
}
