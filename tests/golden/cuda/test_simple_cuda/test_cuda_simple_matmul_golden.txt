// gemm_test(
//     M : size,
//     N : size,
//     K : size,
//     A_cpu : f32[M, K] @DRAM,
//     B_cpu : f32[N, K] @DRAM,
//     C_cpu : f32[N, M] @DRAM
// )
void gemm_test( void *ctxt, int_fast32_t M, int_fast32_t N, int_fast32_t K, float* A_cpu, float* B_cpu, float* C_cpu ) {
  float *A = (float*) exo_cudaMallocAsync(M * K * sizeof(*A), exo_cudaStream);
  float *B = (float*) exo_cudaMallocAsync(N * K * sizeof(*B), exo_cudaStream);
  float *C = (float*) exo_cudaMallocAsync(N * M * sizeof(*C), exo_cudaStream);
  {
    struct exo_CudaDeviceArgs0_gemm_test exo_deviceArgs = {
      M, K, A, exo_excut_get_device_log()
    };
    exo_cudaLaunch0_gemm_test(exo_cudaStream, &exo_deviceArgs);
  }
  {
    struct exo_CudaDeviceArgs1_gemm_test exo_deviceArgs = {
      N, K, B, exo_excut_get_device_log()
    };
    exo_cudaLaunch1_gemm_test(exo_cudaStream, &exo_deviceArgs);
  }
  {
    struct exo_CudaDeviceArgs2_gemm_test exo_deviceArgs = {
      M, N, K, A, B, C, exo_excut_get_device_log()
    };
    exo_cudaLaunch2_gemm_test(exo_cudaStream, &exo_deviceArgs);
  }
  cudaMemcpyAsync(
    (&A_cpu[0]),
    (&A[0]),
    M * K * 4,
    cudaMemcpyDeviceToHost,
    exo_cudaStream);
  exo_cudaFreeAsync(A, exo_cudaStream);
  cudaMemcpyAsync(
    (&B_cpu[0]),
    (&B[0]),
    N * K * 4,
    cudaMemcpyDeviceToHost,
    exo_cudaStream);
  exo_cudaFreeAsync(B, exo_cudaStream);
  cudaMemcpyAsync(
    (&C_cpu[0]),
    (&C[0]),
    N * M * 4,
    cudaMemcpyDeviceToHost,
    exo_cudaStream);
  exo_cudaFreeAsync(C, exo_cudaStream);
}

namespace exo_CudaUtil_test_cuda_simple_matmul_golden {
namespace exo_CudaUtil = ::exo_CudaUtil_test_cuda_simple_matmul_golden;
/* Required by gemm_init_pcg3d_mod(M,K,A,m,k,seed=1337, modulus=7) */
/* Required by gemm_init_pcg3d_mod(M,K,A,m,k,seed=42, modulus=14) */
__device__ void gemm_init_pcg3d_mod(
        uint32_t K, float* A,
        uint32_t m, uint32_t k, uint32_t seed, uint32_t modulus)
{
    uint32_t x = m;
    uint32_t y = k;
    uint32_t z = seed;

    // Copied pseudo random number generation code.
    // http://www.jcgt.org/published/0009/03/02/
    // Hash Functions for GPU Rendering, Mark Jarzynski, Marc Olano, NVIDIA
    x = x*1664525u + 1013904223u;
    y = y*1664525u + 1013904223u;
    z = z*1664525u + 1013904223u;

    x += y*z;
    y += z*x;
    z += x*y;

    x ^= x >> 16u;
    y ^= y >> 16u;
    z ^= z >> 16u;

    x += y*z;
    y += z*x;
    z += x*y;

    float value = (float)((x + y + z) % modulus);

    *A = value;
}

// We need this inline namespace to avoid ODR problems in pytest.
inline namespace exo_CudaInline_test_cuda_simple_matmul_golden {
struct exo_Cuda0_gemm_test
{
  using exo_DeviceArgs = exo_CudaDeviceArgs0_gemm_test;

  static constexpr uint32_t exo_blockDim = 32;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 0;


  struct exo_Task
  {
    int_fast32_t m;
    int_fast32_t k_task;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_m;
    uint32_t exo_cudaTasksNum_m;
    int_fast32_t exo_cudaTasksLo_k_task;
    uint32_t exo_cudaTasksNum_k_task;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_m, int_fast32_t _exo_cudaTasksHi_m,
        int_fast32_t _exo_cudaTasksLo_k_task, int_fast32_t _exo_cudaTasksHi_k_task,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_m = _exo_cudaTasksLo_m;
      exo_cudaTasksNum_m = static_cast<uint32_t>(_exo_cudaTasksHi_m - _exo_cudaTasksLo_m);
      exo_taskCount *= exo_cudaTasksNum_m;
      exo_cudaTasksLo_k_task = _exo_cudaTasksLo_k_task;
      exo_cudaTasksNum_k_task = static_cast<uint32_t>(_exo_cudaTasksHi_k_task - _exo_cudaTasksLo_k_task);
      exo_taskCount *= exo_cudaTasksNum_k_task;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.k_task = exo_cudaTasksLo_k_task + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_k_task);
      exo_tmp /= exo_cudaTasksNum_k_task;
      exo_task.m = exo_cudaTasksLo_m + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_m);
      exo_tmp /= exo_cudaTasksNum_m;
      return exo_task;
    }
  };

  struct exo_SyncState
  {

  };

  static inline const char*& exo_FILE()
  {
    static const char* name = __FILE__;
    return name;
  }

  static void
  exo_cudaLaunch(cudaStream_t exo_cudaStream, const exo_DeviceArgs& exo_deviceArgs);

  static __device__ __forceinline__ void
  exo_deviceSetup(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceMainLoop(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceTask(
      char* exo_smem,
      exo_SyncState& exo_syncState,
      const exo_DeviceArgs& exo_deviceArgs,
      exo_Task exo_task,
      exo_ExcutThreadLog exo_excutLog={});
};
}  // end inline namespace

inline void
exo_CudaInline_test_cuda_simple_matmul_golden::exo_Cuda0_gemm_test::exo_cudaLaunch(
    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_cuda_simple_matmul_golden;
  cudaFuncSetAttribute(exo_deviceFunction0_gemm_test, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 1u;

  cudaLaunchConfig_t exo_launchConfig = {};
  exo_launchConfig.gridDim = dim3(exo_gridDim, 1, 1);
  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

  cudaLaunchKernelEx(&exo_launchConfig, exo_deviceFunction0_gemm_test, exo_deviceArgs);

  exo_excut_flush_device_log(
      exo_cudaStream, exo_gridDim, exo_blockDim,
      exo_CudaUtil::exo_excut_str_id_count, exo_CudaUtil::exo_excut_str_table,
      1, &exo_FILE());
}

__device__ __forceinline__ void
exo_CudaInline_test_cuda_simple_matmul_golden::exo_Cuda0_gemm_test::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_cuda_simple_matmul_golden;
  // cuda_threads(0, 32, unit=cuda_thread)
  if ([[maybe_unused]] int exo_1thr_k_seq = threadIdx.x; 1) {
    exo_CudaUtil::gemm_init_pcg3d_mod(exo_deviceArgs.K, &exo_deviceArgs.A[exo_task.m * exo_deviceArgs.K + (exo_task.k_task * 32 + exo_1thr_k_seq)], exo_task.m, (exo_task.k_task * 32 + exo_1thr_k_seq), 1337, 7);
  }
}
__device__ __forceinline__ void
exo_CudaInline_test_cuda_simple_matmul_golden::exo_Cuda0_gemm_test::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_cuda_simple_matmul_golden;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 32) {
      {
        int_fast32_t exo_cudaTasksLo_m = 0;
        int_fast32_t exo_cudaTasksHi_m = exo_deviceArgs.M;
        {
          int_fast32_t exo_cudaTasksLo_k_task = 0;
          int_fast32_t exo_cudaTasksHi_k_task = ((exo_deviceArgs.K) / (32));
          exo_TaskGenerator exo_taskGenerator(
              blockIdx.x / exo_clusterDim,
              gridDim.x / exo_clusterDim,
              exo_cudaTasksLo_m, exo_cudaTasksHi_m,

__device__ __forceinline__ void
exo_CudaInline_test_cuda_simple_matmul_golden::exo_Cuda2_gemm_test::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_cuda_simple_matmul_golden;
  auto& A_smem = reinterpret_cast<float (&) [4096]>(exo_smem[exo_smemOffset0_A_smem]);
  auto& B_smem = reinterpret_cast<float (&) [8192]>(exo_smem[exo_smemOffset1_B_smem]);
  float accum[8 * 16];
  // cuda_threads(0, 16, unit=16 * cuda_thread)
  if ([[maybe_unused]] int exo_16thr_m1 = (threadIdx.x / 16); 1) {
    // cuda_threads(0, 16, unit=cuda_thread)
    if ([[maybe_unused]] int exo_1thr_n1 = (threadIdx.x % 16); 1) {
      for (int m0 = 0; m0 < 8; m0++) {
        for (int n0 = 0; n0 < 16; n0++) {
          accum[m0 * 16 + n0] = 0.0f;
        }
      }
    }
  }
  for (int k1 = 0; k1 < ((exo_deviceArgs.K) / (32)); k1++) {
    for (int m1 = 0; m1 < 16; m1++) {
      // cuda_threads(0, 8, unit=32 * cuda_thread)
      if ([[maybe_unused]] int exo_32thr_m0 = (threadIdx.x / 32); 1) {
        // cuda_threads(0, 32, unit=cuda_thread)
        if ([[maybe_unused]] int exo_1thr_k0 = (threadIdx.x % 32); 1) {
          A_smem[(m1 * 8 + exo_32thr_m0) * 32 + exo_1thr_k0] = exo_deviceArgs.A[(exo_task.m2 * 128 + m1 * 8 + exo_32thr_m0) * exo_deviceArgs.K + (k1 * 32 + exo_1thr_k0)];
        }
      }
    }
    for (int n1 = 0; n1 < 32; n1++) {
      // cuda_threads(0, 8, unit=32 * cuda_thread)
      if ([[maybe_unused]] int exo_32thr_n0 = (threadIdx.x / 32); 1) {
        // cuda_threads(0, 32, unit=cuda_thread)
        if ([[maybe_unused]] int exo_1thr_k0 = (threadIdx.x % 32); 1) {
          B_smem[(n1 * 8 + exo_32thr_n0) * 32 + exo_1thr_k0] = exo_deviceArgs.B[(exo_task.n2 * 256 + n1 * 8 + exo_32thr_n0) * exo_deviceArgs.K + (k1 * 32 + exo_1thr_k0)];
        }
      }
    }
    // Fence(cuda_in_order, cuda_in_order)
    asm volatile(
      "barrier.cta.sync 0;"
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
    exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
    // cuda_threads(0, 16, unit=16 * cuda_thread)
    if ([[maybe_unused]] int exo_16thr_m1 = (threadIdx.x / 16); 1) {
      // cuda_threads(0, 16, unit=cuda_thread)
      if ([[maybe_unused]] int exo_1thr_n1 = (threadIdx.x % 16); 1) {
        for (int k0 = 0; k0 < 32; k0++) {
          for (int m0 = 0; m0 < 8; m0++) {
            for (int n0 = 0; n0 < 16; n0++) {
              accum[m0 * 16 + n0] += A_smem[(exo_16thr_m1 * 8 + m0) * 32 + k0] * B_smem[(exo_1thr_n1 * 16 + n0) * 32 + k0];
            }
          }
        }
      }
    }
    // Fence(cuda_in_order, cuda_in_order)
    asm volatile(
      "barrier.cta.sync 0;"
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
    exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
  }
  // cuda_threads(0, 16, unit=16 * cuda_thread)
  if ([[maybe_unused]] int exo_16thr_m1 = (threadIdx.x / 16); 1) {
    // cuda_threads(0, 16, unit=cuda_thread)
    if ([[maybe_unused]] int exo_1thr_n1 = (threadIdx.x % 16); 1) {
      for (int m0 = 0; m0 < 8; m0++) {
        for (int n0 = 0; n0 < 16; n0++) {
          exo_deviceArgs.C[(exo_task.n2 * 256 + exo_1thr_n1 * 16 + n0) * exo_deviceArgs.M + (exo_task.m2 * 128 + exo_16thr_m1 * 8 + m0)] = accum[m0 * 16 + n0];
        }
      }
    }
  }
}
