// gemv_warp_coop_8(
//     M : size,
//     K : size,
//     h_A : f32[M, K] @DRAM,
//     h_x : f32[K] @DRAM,
//     h_y : f32[M] @DRAM
// )
void gemv_warp_coop_8( void *ctxt, int_fast32_t M, int_fast32_t K, const float* h_A, const float* h_x, float* h_y ) {
EXO_ASSUME(M % 16 == 0);
EXO_ASSUME(K % 8 == 0);
EXO_ASSUME(K <= 2048);
float *d_A = (float*) exo_cudaMallocAsync(M * K * sizeof(*d_A), exo_cudaStream);
float *d_x = (float*) exo_cudaMallocAsync(K * sizeof(*d_x), exo_cudaStream);
float *d_y = (float*) exo_cudaMallocAsync(M * sizeof(*d_y), exo_cudaStream);
cudaMemcpyAsync(&d_A[0], &h_A[0], 4 * (M) * (K), cudaMemcpyHostToDevice, exo_cudaStream);
cudaMemcpyAsync(&d_x[0], &h_x[0], 4 * (K), cudaMemcpyHostToDevice, exo_cudaStream);
{
  struct exo_CudaDeviceArgs0_gemv_warp_coop_8 exo_deviceArgs = {
    M, K, d_A, d_x, d_y, exo_excut_get_device_log()
  };
  exo_cudaLaunch0_gemv_warp_coop_8(exo_cudaStream, &exo_deviceArgs);
}
exo_cudaFreeAsync(d_x, exo_cudaStream);
exo_cudaFreeAsync(d_A, exo_cudaStream);
cudaMemcpyAsync(&h_y[0], &d_y[0], 4 * (M), cudaMemcpyDeviceToHost, exo_cudaStream);
exo_cudaFreeAsync(d_y, exo_cudaStream);
}

  struct exo_Task
  {
    int_fast32_t m1;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_m1;
    uint32_t exo_cudaTasksNum_m1;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_m1, int_fast32_t _exo_cudaTasksHi_m1,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_m1 = _exo_cudaTasksLo_m1;
      exo_cudaTasksNum_m1 = static_cast<uint32_t>(_exo_cudaTasksHi_m1 - _exo_cudaTasksLo_m1);
      exo_taskCount *= exo_cudaTasksNum_m1;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.m1 = exo_cudaTasksLo_m1 + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_m1);
      exo_tmp /= exo_cudaTasksNum_m1;
      return exo_task;
    }
  };

  };

    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_gemv_warp_coop_8_golden;
  cudaFuncSetAttribute(exo_deviceFunction0_gemv_warp_coop_8, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 1u;

  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

__device__ __forceinline__ void
exo_CudaInline_test_gemv_warp_coop_8_golden::exo_Cuda0_gemv_warp_coop_8::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_gemv_warp_coop_8_golden;
  auto& shared_x = reinterpret_cast<float (&) [2048]>(exo_smem[exo_smemOffset0_shared_x]);
  for (int i = 0; i < ((exo_deviceArgs.K + 128 - 1) / (128)); i++) {
    // cuda_threads(0, 128, unit=cuda_thread)
    if ([[maybe_unused]] int exo_1thr_tid = threadIdx.x; 1) {
      if (i * 128 + exo_1thr_tid < exo_deviceArgs.K) {
        shared_x[i * 128 + exo_1thr_tid] = exo_deviceArgs.d_x[i * 128 + exo_1thr_tid];
      }
    }
  }
  // Fence(cuda_in_order, cuda_in_order)
  asm volatile(
    "barrier.cta.sync 0;"
  );
  exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
  exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
  // cuda_threads(0, 16, unit=8 * cuda_thread)
  if ([[maybe_unused]] int exo_8thr_m0 = (threadIdx.x / 8); 1) {
    float partial_sum;
    // cuda_threads(0, 2, unit=4 * cuda_thread)
    if ([[maybe_unused]] int exo_4thr_k4 = (threadIdx.x % 8 / 4); 1) {
      // cuda_threads(0, 2, unit=2 * cuda_thread)
      if ([[maybe_unused]] int exo_2thr_k2 = (threadIdx.x % 4 / 2); 1) {
        // cuda_threads(0, 2, unit=cuda_thread)
        if ([[maybe_unused]] int exo_1thr_k1 = (threadIdx.x % 2); 1) {
          partial_sum = 0.0f;
          for (int k8 = 0; k8 < ((exo_deviceArgs.K) / (8)); k8++) {
            partial_sum += exo_deviceArgs.d_A[(exo_task.m1 * 16 + exo_8thr_m0) * (exo_deviceArgs.K) + k8 * 8 + exo_4thr_k4 * 4 + exo_2thr_k2 * 2 + exo_1thr_k1] * shared_x[k8 * 8 + exo_4thr_k4 * 4 + exo_2thr_k2 * 2 + exo_1thr_k1];
          }
        }
      }
    }
    float tmp_1;
    // cuda_threads(0, 2, unit=2 * cuda_threads_strided(2, 4))
    if ([[maybe_unused]] int exo_2thr_k2 = (threadIdx.x % 4 / 2); 1) {
      // cuda_threads(0, 2, unit=2 * cuda_threads_strided(1, 4))
      if ([[maybe_unused]] int exo_1thr_k1 = (threadIdx.x % 2); 1) {
        tmp_1 = partial_sum + __shfl_xor_sync(17 << (threadIdx.x & 27), partial_sum, 4);
      }
    }
    // cuda_threads(0, 2, unit=4 * cuda_thread)
    if ([[maybe_unused]] int exo_4thr_k4 = (threadIdx.x % 8 / 4); 1) {
      // cuda_threads(0, 2, unit=2 * cuda_threads_strided(1, 2))
      if ([[maybe_unused]] int exo_1thr_k1 = (threadIdx.x % 2); 1) {
        partial_sum = tmp_1 + __shfl_xor_sync(5 << (threadIdx.x & 29), tmp_1, 2);
      }
    }
    // cuda_threads(0, 2, unit=4 * cuda_thread)
    if ([[maybe_unused]] int exo_4thr_k4 = (threadIdx.x % 8 / 4); 1) {
      // cuda_threads(0, 2, unit=2 * cuda_thread)
      if ([[maybe_unused]] int exo_2thr_k2 = (threadIdx.x % 4 / 2); 1) {
        tmp_1 = partial_sum + __shfl_xor_sync(3 << (threadIdx.x & 30), partial_sum, 1);
        // cuda_threads(0, 2, unit=cuda_thread)
        if ([[maybe_unused]] int exo_1thr_k1 = (threadIdx.x % 2); 1) {
          if (exo_4thr_k4 == 0) {
            if (exo_2thr_k2 == 0) {
              if (exo_1thr_k1 == 0) {
                exo_deviceArgs.d_y[exo_task.m1 * 16 + exo_8thr_m0] = tmp_1;
              }
            }
          }
        }
      }
    }
  }
  // Fence(cuda_in_order, cuda_in_order)
  asm volatile(
    "barrier.cta.sync 0;"
  );
  exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
  exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
}
