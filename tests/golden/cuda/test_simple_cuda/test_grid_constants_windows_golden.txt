// weird_windows(
//     N : size,
//     test_scalar : i32 @DRAM,
//     test_vector : i32[8] @DRAM,
//     test_out : i32[N, N] @DRAM
// )
void weird_windows( void *ctxt, int_fast32_t N, const int32_t* test_scalar, const int32_t* test_vector, int32_t* test_out ) {
  EXO_ASSUME(N > 4);
  int32_t *test_mem = (int32_t*) exo_cudaMallocAsync(N * N * sizeof(*test_mem), exo_cudaStream);
  int32_t const_scalar;
  const_scalar = *test_scalar;
  int32_t const_vector[4];
  for (int_fast32_t i = 0; i < 4; i++) {
    const_vector[i] = test_vector[i];
  }
  int32_t *test_vector_gmem = (int32_t*) exo_cudaMallocAsync(8 * sizeof(*test_vector_gmem), exo_cudaStream);
  cudaMemcpyAsync(&test_vector_gmem[0], &test_vector[0], 4 * 8, cudaMemcpyHostToDevice, exo_cudaStream);
  struct exo_win_1i32 input_window = (struct exo_win_1i32) { &test_vector_gmem[4], {1} };
  for (int_fast32_t i = 0; i < N; i++) {
    for (int_fast32_t N_1 = 0; N_1 < 4; N_1++) {
      {
        struct exo_CudaDeviceArgs0_weird_windows exo_deviceArgs = {
          N, test_mem, const_scalar, {const_vector[0], const_vector[1], const_vector[2], const_vector[3]}, (struct exo_win_1i32) { &input_window.data[0], {input_window.strides[0]} }, i, N_1, exo_excut_get_device_log()
        };
        exo_cudaLaunch0_weird_windows(exo_cudaStream, &exo_deviceArgs);
      }
    }
  }
  for (int_fast32_t col = 4; col < N; col++) {
    struct exo_win_1i32 col_window = (struct exo_win_1i32) { &test_mem[col], {N} };
    {
      struct exo_CudaDeviceArgs1_weird_windows exo_deviceArgs = {
        N, {const_vector[0], const_vector[1], const_vector[2], const_vector[3]}, test_vector_gmem, (struct exo_win_1i32) { &input_window.data[0], {input_window.strides[0]} }, col, (struct exo_win_1i32) { &col_window.data[0], {col_window.strides[0]} }, exo_excut_get_device_log()
      };
      exo_cudaLaunch1_weird_windows(exo_cudaStream, &exo_deviceArgs);
    }
  }
  exo_cudaFreeAsync(test_vector_gmem, exo_cudaStream);
  cudaMemcpyAsync(&test_out[0], &test_mem[0], 4 * N * N, cudaMemcpyDeviceToHost, exo_cudaStream);
  exo_cudaFreeAsync(test_mem, exo_cudaStream);
}

// We need this inline namespace to avoid ODR problems in pytest.
inline namespace exo_CudaInline_test_grid_constants_windows_golden {
struct exo_Cuda0_weird_windows
{
  using exo_DeviceArgs = exo_CudaDeviceArgs0_weird_windows;

  static constexpr uint32_t exo_blockDim = 32;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 0;


  struct exo_Task
  {
    int_fast32_t task;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_task;
    uint32_t exo_cudaTasksNum_task;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_task, int_fast32_t _exo_cudaTasksHi_task,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_task = _exo_cudaTasksLo_task;
      exo_cudaTasksNum_task = static_cast<uint32_t>(_exo_cudaTasksHi_task - _exo_cudaTasksLo_task);
      exo_taskCount *= exo_cudaTasksNum_task;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.task = exo_cudaTasksLo_task + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_task);
      exo_tmp /= exo_cudaTasksNum_task;
      return exo_task;
    }
  };

  struct exo_SyncState
  {

  };

  static inline const char*& exo_FILE()
  {
    static const char* name = __FILE__;
    return name;
  }

  static void
  exo_cudaLaunch(cudaStream_t exo_cudaStream, const exo_DeviceArgs& exo_deviceArgs);

  static __device__ __forceinline__ void
  exo_deviceSetup(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceMainLoop(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceTask(
      char* exo_smem,
      exo_SyncState& exo_syncState,
      const exo_DeviceArgs& exo_deviceArgs,
      exo_Task exo_task,
      exo_ExcutThreadLog exo_excutLog={});
};
}  // end inline namespace

inline void
exo_CudaInline_test_grid_constants_windows_golden::exo_Cuda0_weird_windows::exo_cudaLaunch(
    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_grid_constants_windows_golden;
  cudaFuncSetAttribute(exo_deviceFunction0_weird_windows, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 1u;

  cudaLaunchConfig_t exo_launchConfig = {};
  exo_launchConfig.gridDim = dim3(exo_gridDim, 1, 1);
  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

  cudaLaunchKernelEx(&exo_launchConfig, exo_deviceFunction0_weird_windows, exo_deviceArgs);

  exo_excut_flush_device_log(
      exo_cudaStream, exo_gridDim, exo_blockDim,
      exo_CudaUtil::exo_excut_str_id_count, exo_CudaUtil::exo_excut_str_table,
      1, &exo_FILE());
}

    char* exo_smem,

  // No mbarriers used

__device__ __forceinline__ void
exo_CudaInline_test_grid_constants_windows_golden::exo_Cuda0_weird_windows::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_grid_constants_windows_golden;
  // cuda_threads(0, 1, unit=cuda_thread)
  if (int exo_0thr_tid = threadIdx.x; exo_0thr_tid < 1) {
    struct exo_win_1i32c const_vector_window = (struct exo_win_1i32c) { &exo_deviceArgs.const_vector[exo_deviceArgs.N_1], {1} };
    exo_deviceArgs.test_mem[exo_deviceArgs.i * exo_deviceArgs.N + exo_deviceArgs.N_1] = exo_deviceArgs.const_scalar * const_vector_window.data[0] * exo_deviceArgs.input_window.data[exo_deviceArgs.N_1 * exo_deviceArgs.input_window.strides[0]];
  }
}
__device__ __forceinline__ void
exo_CudaInline_test_grid_constants_windows_golden::exo_Cuda0_weird_windows::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_grid_constants_windows_golden;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 32) {
      {
        int_fast32_t exo_cudaTasksLo_task = 0;
        int_fast32_t exo_cudaTasksHi_task = 1;
        exo_TaskGenerator exo_taskGenerator(
            blockIdx.x / exo_clusterDim,
            gridDim.x / exo_clusterDim,
            exo_cudaTasksLo_task, exo_cudaTasksHi_task,
            exo_deviceArgs);
        while (exo_taskGenerator.prepare_next_task()) {
          exo_deviceTask(exo_smem, exo_syncState, exo_deviceArgs, exo_taskGenerator.get_next_task(), exo_excutLog);
        }
      }
    }
  }
}

    int32_t const_vector[4];  // const_vector: i32[4] @CudaGridConstant
    int32_t* test_vector_gmem;  // test_vector_gmem : i32[8] @CudaGmemLinear
    struct exo_win_1i32 input_window;  // input_window : Window(src_type=i32[8], as_tensor=[i32][8 - 4], src_buf=test_vector_gmem, idx='[4:8]') @CudaGmemLinear
    int_fast32_t col;  // col : index
    struct exo_win_1i32 col_window;  // col_window : Window(src_type=i32[N, N], as_tensor=[i32][N], src_buf=test_mem, idx='[0:N, col]') @CudaGmemLinear
    EXO_EXCUT_DEVICE_LOG_MEMBER  // for Exo pytest (exo_excut.h)
};

// We need this inline namespace to avoid ODR problems in pytest.
inline namespace exo_CudaInline_test_grid_constants_windows_golden {
struct exo_Cuda1_weird_windows
{
  using exo_DeviceArgs = exo_CudaDeviceArgs1_weird_windows;

  static constexpr uint32_t exo_blockDim = 32;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 0;


  struct exo_Task
  {
    int_fast32_t task;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_task;
    uint32_t exo_cudaTasksNum_task;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_task, int_fast32_t _exo_cudaTasksHi_task,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_task = _exo_cudaTasksLo_task;
      exo_cudaTasksNum_task = static_cast<uint32_t>(_exo_cudaTasksHi_task - _exo_cudaTasksLo_task);
      exo_taskCount *= exo_cudaTasksNum_task;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.task = exo_cudaTasksLo_task + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_task);
      exo_tmp /= exo_cudaTasksNum_task;
      return exo_task;
    }
  };

  struct exo_SyncState
  {

  };

  static inline const char*& exo_FILE()
  {
    static const char* name = __FILE__;
    return name;
  }

  static void
  exo_cudaLaunch(cudaStream_t exo_cudaStream, const exo_DeviceArgs& exo_deviceArgs);

  static __device__ __forceinline__ void
  exo_deviceSetup(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceMainLoop(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceTask(
      char* exo_smem,
      exo_SyncState& exo_syncState,
      const exo_DeviceArgs& exo_deviceArgs,
      exo_Task exo_task,
      exo_ExcutThreadLog exo_excutLog={});
};
}  // end inline namespace

inline void
exo_CudaInline_test_grid_constants_windows_golden::exo_Cuda1_weird_windows::exo_cudaLaunch(
    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_grid_constants_windows_golden;
  cudaFuncSetAttribute(exo_deviceFunction1_weird_windows, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 1u;

  cudaLaunchConfig_t exo_launchConfig = {};
  exo_launchConfig.gridDim = dim3(exo_gridDim, 1, 1);
  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

  cudaLaunchKernelEx(&exo_launchConfig, exo_deviceFunction1_weird_windows, exo_deviceArgs);

  exo_excut_flush_device_log(
      exo_cudaStream, exo_gridDim, exo_blockDim,
      exo_CudaUtil::exo_excut_str_id_count, exo_CudaUtil::exo_excut_str_table,
      1, &exo_FILE());
}

__device__ __forceinline__ void
exo_CudaInline_test_grid_constants_windows_golden::exo_Cuda1_weird_windows::exo_deviceSetup(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  // No mbarriers used
}

__device__ __forceinline__ void
exo_CudaInline_test_grid_constants_windows_golden::exo_Cuda1_weird_windows::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_grid_constants_windows_golden;
  // cuda_threads(0, 1, unit=cuda_thread)
  if (int exo_0thr_tid = threadIdx.x; exo_0thr_tid < 1) {
    exo_deviceArgs.col_window.data[exo_task.task * exo_deviceArgs.col_window.strides[0]] = exo_deviceArgs.input_window.data[(exo_task.task & 3) * exo_deviceArgs.input_window.strides[0]] * exo_deviceArgs.const_vector[(exo_deviceArgs.col & 3)] + exo_deviceArgs.test_vector_gmem[(exo_task.task & 3)];
  }
}
__device__ __forceinline__ void
exo_CudaInline_test_grid_constants_windows_golden::exo_Cuda1_weird_windows::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_grid_constants_windows_golden;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 32) {
      {
        int_fast32_t exo_cudaTasksLo_task = 0;
        int_fast32_t exo_cudaTasksHi_task = exo_deviceArgs.N;
        exo_TaskGenerator exo_taskGenerator(
            blockIdx.x / exo_clusterDim,
            gridDim.x / exo_clusterDim,
            exo_cudaTasksLo_task, exo_cudaTasksHi_task,
            exo_deviceArgs);
        while (exo_taskGenerator.prepare_next_task()) {
          exo_deviceTask(exo_smem, exo_syncState, exo_deviceArgs, exo_taskGenerator.get_next_task(), exo_excutLog);
        }
      }
    }
  }
}
