// array_test(
//     exo_user_exo_ZZ0a : size,
//     exo_user_exo_ZZ0b : size,
//     exo_user_exo_ZZ1a : size,
//     exo_user_exo_ZZ1b : size,
//     exo_user_exo_ZZ2 : i32[exo_ZZ0a + exo_ZZ0b, exo_ZZ1a + exo_ZZ1b] @DRAM,
//     exo_user_exo_ZZ3 : f32[4] @DRAM,
//     exo_user_exo_ZZ4 : f32[4] @DRAM
// )
void array_test( void *ctxt, int_fast32_t exo_user_exo_ZZ0a, int_fast32_t exo_user_exo_ZZ0b, int_fast32_t exo_user_exo_ZZ1a, int_fast32_t exo_user_exo_ZZ1b, int32_t* exo_user_exo_ZZ2, const float* exo_user_exo_ZZ3, const float* exo_user_exo_ZZ4 ) {
  EXO_ASSUME(exo_user_exo_ZZ0b > 4);
  EXO_ASSUME(exo_user_exo_ZZ1b > 4);
  int32_t *exo_user_exo_ZZ2_device = (int32_t*) exo_cudaMallocAsync((exo_user_exo_ZZ0a + exo_user_exo_ZZ0b) * (exo_user_exo_ZZ1a + exo_user_exo_ZZ1b) * sizeof(*exo_user_exo_ZZ2_device), exo_cudaStream);
  float exo_user_exo_ZZ3_device[4];
  float exo_user_exo_ZZ4_device[4];
  for (int_fast32_t i = 0; i < 4; i++) {
    exo_user_exo_ZZ3_device[i] = exo_user_exo_ZZ3[i];
    exo_user_exo_ZZ4_device[i] = exo_user_exo_ZZ4[i];
  }
  {
    struct exo_CudaDeviceArgs0_array_test exo_deviceArgs = {
      exo_user_exo_ZZ0a, exo_user_exo_ZZ0b, exo_user_exo_ZZ1a, exo_user_exo_ZZ1b, exo_user_exo_ZZ2_device, {exo_user_exo_ZZ3_device[0], exo_user_exo_ZZ3_device[1], exo_user_exo_ZZ3_device[2], exo_user_exo_ZZ3_device[3]}, {exo_user_exo_ZZ4_device[0], exo_user_exo_ZZ4_device[1], exo_user_exo_ZZ4_device[2], exo_user_exo_ZZ4_device[3]}, exo_excut_get_device_log()
    };
    exo_cudaLaunch0_array_test(exo_cudaStream, &exo_deviceArgs);
  }
  cudaMemcpyAsync(
    (&exo_user_exo_ZZ2[0]),
    (&exo_user_exo_ZZ2_device[0]),
    (exo_user_exo_ZZ0a + exo_user_exo_ZZ0b) * (exo_user_exo_ZZ1a + exo_user_exo_ZZ1b) * 4,
    cudaMemcpyDeviceToHost,
    exo_cudaStream);
  exo_cudaFreeAsync(exo_user_exo_ZZ2_device, exo_cudaStream);
}

// We need this inline namespace to avoid ODR problems in pytest.
inline namespace exo_CudaInline_test_cuda_arrays_golden {
struct exo_Cuda0_array_test
{
  using exo_DeviceArgs = exo_CudaDeviceArgs0_array_test;

  static constexpr uint32_t exo_blockDim = 32;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 0;


  struct exo_Task
  {
    int_fast32_t task;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_task;
    uint32_t exo_cudaTasksNum_task;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_task, int_fast32_t _exo_cudaTasksHi_task,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_task = _exo_cudaTasksLo_task;
      exo_cudaTasksNum_task = static_cast<uint32_t>(_exo_cudaTasksHi_task - _exo_cudaTasksLo_task);
      exo_taskCount *= exo_cudaTasksNum_task;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.task = exo_cudaTasksLo_task + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_task);
      exo_tmp /= exo_cudaTasksNum_task;
      return exo_task;
    }
  };

  struct exo_SyncState
  {

  };

  static inline const char*& exo_FILE()
  {
    static const char* name = __FILE__;
    return name;
  }

  static void
  exo_cudaLaunch(cudaStream_t exo_cudaStream, const exo_DeviceArgs& exo_deviceArgs);

  static __device__ __forceinline__ void
  exo_deviceSetup(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceShutdown(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceMainLoop(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceTask(
      char* exo_smem,
      exo_SyncState& exo_syncState,
      const exo_DeviceArgs& exo_deviceArgs,
      exo_Task exo_task,
      exo_ExcutThreadLog exo_excutLog={});
};
}  // end inline namespace

inline void
exo_CudaInline_test_cuda_arrays_golden::exo_Cuda0_array_test::exo_cudaLaunch(
    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_cuda_arrays_golden;
  cudaFuncSetAttribute(exo_deviceFunction0_array_test, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 1u;

  cudaLaunchConfig_t exo_launchConfig = {};
  exo_launchConfig.gridDim = dim3(exo_gridDim, 1, 1);
  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

  cudaLaunchKernelEx(&exo_launchConfig, exo_deviceFunction0_array_test, exo_deviceArgs);

  exo_excut_flush_device_log(
      exo_cudaStream, exo_gridDim, exo_blockDim,
      exo_CudaUtil::exo_excut_str_id_count, exo_CudaUtil::exo_excut_str_table,
      1, &exo_FILE());
}

__device__ __forceinline__ void
exo_CudaInline_test_cuda_arrays_golden::exo_Cuda0_array_test::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_cuda_arrays_golden;
  // cuda_threads(0, 4, unit=cuda_thread)
  if (int exo_1thr_tid = threadIdx.x; exo_1thr_tid < 4) {
    exo_deviceArgs.exo_user_exo_ZZ2_device[(exo_deviceArgs.exo_user_exo_ZZ0a + exo_1thr_tid) * (exo_deviceArgs.exo_user_exo_ZZ1a + exo_deviceArgs.exo_user_exo_ZZ1b) + (exo_deviceArgs.exo_user_exo_ZZ1a + 4 - exo_1thr_tid)] = (int32_t)(exo_deviceArgs.exo_user_exo_ZZ3_device[exo_1thr_tid] + exo_deviceArgs.exo_user_exo_ZZ4_device[exo_1thr_tid]);
  }
}
__device__ __forceinline__ void
exo_CudaInline_test_cuda_arrays_golden::exo_Cuda0_array_test::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_cuda_arrays_golden;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 32) {
      {
        int_fast32_t exo_cudaTasksLo_task = 0;
        int_fast32_t exo_cudaTasksHi_task = 1;
        exo_TaskGenerator exo_taskGenerator(
            blockIdx.x / exo_clusterDim,
            gridDim.x / exo_clusterDim,
            exo_cudaTasksLo_task, exo_cudaTasksHi_task,
            exo_deviceArgs);
        while (exo_taskGenerator.prepare_next_task()) {
          exo_deviceTask(exo_smem, exo_syncState, exo_deviceArgs, exo_taskGenerator.get_next_task(), exo_excutLog);
        }
      }
    }
  }
}
