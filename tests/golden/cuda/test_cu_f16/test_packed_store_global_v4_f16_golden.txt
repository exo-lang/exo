template <typename PtxType, typename Scalar, typename PackedStruct>
struct exo_CudaRmemPacked32
{
    static_assert(sizeof(PtxType) == sizeof(PackedStruct));
    PtxType ptx_data;

    template <typename Index>
    __device__ auto operator[] (Index i) const -> Scalar
    {
        if constexpr (sizeof(Scalar) == 4)
            return ptx_data;
        else if (i == 0)
            return reinterpret_cast<const PackedStruct*>(&ptx_data)->x;
        else
            return reinterpret_cast<const PackedStruct*>(&ptx_data)->y;
    }
};

using exo_CudaRmemPacked32_f32 = exo_CudaRmemPacked32<float, float, float>;
using exo_CudaRmemPacked32_i32 = exo_CudaRmemPacked32<int32_t, int32_t, int32_t>;

using exo_CudaRmemPacked32_bf16 = exo_CudaRmemPacked32<int32_t, __nv_bfloat16, __nv_bfloat162>;

// foo(
//     h_dst : f16[64] @DRAM,
//     h_src : f16[64] @DRAM
// )
void foo( void *ctxt, exo_f16* h_dst, const exo_f16* h_src ) {
  exo_f16 *d_dst = (exo_f16*) exo_cudaMallocAsync(64 * sizeof(*d_dst), exo_cudaStream);
  exo_f16 *d_src = (exo_f16*) exo_cudaMallocAsync(64 * sizeof(*d_src), exo_cudaStream);
  cudaMemcpyAsync(
    (&d_src[0]),
    (&h_src[0]),
    64 * 2,
    cudaMemcpyHostToDevice,
    exo_cudaStream);
  {
    struct exo_CudaDeviceArgs0_foo exo_deviceArgs = {
      d_dst, d_src, exo_excut_get_device_log()
    };
    exo_cudaLaunch0_foo(exo_cudaStream, &exo_deviceArgs);
  }
  exo_cudaFreeAsync(d_src, exo_cudaStream);
  cudaMemcpyAsync(
    (&h_dst[0]),
    (&d_dst[0]),
    64 * 2,
    cudaMemcpyDeviceToHost,
    exo_cudaStream);
  exo_cudaFreeAsync(d_dst, exo_cudaStream);
}

// We need this inline namespace to avoid ODR problems in pytest.
inline namespace exo_CudaInline_test_packed_store_global_v4_f16_golden {
struct exo_Cuda0_foo
{
  using exo_DeviceArgs = exo_CudaDeviceArgs0_foo;

  static constexpr uint32_t exo_blockDim = 32;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 0;


  struct exo_Task
  {
    int_fast32_t task;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_task;
    uint32_t exo_cudaTasksNum_task;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_task, int_fast32_t _exo_cudaTasksHi_task,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_task = _exo_cudaTasksLo_task;
      exo_cudaTasksNum_task = static_cast<uint32_t>(_exo_cudaTasksHi_task - _exo_cudaTasksLo_task);
      exo_taskCount *= exo_cudaTasksNum_task;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.task = exo_cudaTasksLo_task + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_task);
      exo_tmp /= exo_cudaTasksNum_task;
      return exo_task;
    }
  };

  struct exo_SyncState
  {

  };

  static inline const char*& exo_FILE()
  {
    static const char* name = __FILE__;
    return name;
  }

  static void
  exo_cudaLaunch(cudaStream_t exo_cudaStream, const exo_DeviceArgs& exo_deviceArgs);

  static __device__ __forceinline__ void
  exo_deviceSetup(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceMainLoop(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceTask(
      char* exo_smem,
      exo_SyncState& exo_syncState,
      const exo_DeviceArgs& exo_deviceArgs,
      exo_Task exo_task,
      exo_ExcutThreadLog exo_excutLog={});
};
}  // end inline namespace

inline void
exo_CudaInline_test_packed_store_global_v4_f16_golden::exo_Cuda0_foo::exo_cudaLaunch(
    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_packed_store_global_v4_f16_golden;
  cudaFuncSetAttribute(exo_deviceFunction0_foo, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 1u;

  cudaLaunchConfig_t exo_launchConfig = {};
  exo_launchConfig.gridDim = dim3(exo_gridDim, 1, 1);
  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

  cudaLaunchKernelEx(&exo_launchConfig, exo_deviceFunction0_foo, exo_deviceArgs);

  exo_excut_flush_device_log(
      exo_cudaStream, exo_gridDim, exo_blockDim,
      exo_CudaUtil::exo_excut_str_id_count, exo_CudaUtil::exo_excut_str_table,
      1, &exo_FILE());
}

  // No mbarriers used

__device__ __forceinline__ void
exo_CudaInline_test_packed_store_global_v4_f16_golden::exo_Cuda0_foo::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_packed_store_global_v4_f16_golden;
  exo_CudaRmemPacked32_f16 rmem[4];
  // cuda_threads(0, 8, unit=cuda_thread)
  if (int exo_1thr_tid = threadIdx.x; exo_1thr_tid < 8) {
    for (int i = 0; i < 4; i++) {
      rmem[i].ptx_data = *reinterpret_cast<const decltype(rmem[i].ptx_data)*>((&exo_deviceArgs.d_src[(2 * i + 8 * exo_1thr_tid)]));
    }
    asm volatile(
      "st.global.v4.u32 [%0], {%1, %2, %3, %4};"
        :
        :"l"((&exo_deviceArgs.d_dst[(8 * exo_1thr_tid)])),
        "r"(rmem[0].ptx_data), "r"(rmem[1].ptx_data), "r"(rmem[2].ptx_data), "r"(rmem[3].ptx_data)
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(st_global_v4_u32), 0, __LINE__);
    exo_excutLog.log_ptr_arg((&exo_deviceArgs.d_dst[(8 * exo_1thr_tid)]));
  }
}
__device__ __forceinline__ void
exo_CudaInline_test_packed_store_global_v4_f16_golden::exo_Cuda0_foo::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_packed_store_global_v4_f16_golden;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 32) {
      {
        int_fast32_t exo_cudaTasksLo_task = 0;
        int_fast32_t exo_cudaTasksHi_task = 1;
        exo_TaskGenerator exo_taskGenerator(
            blockIdx.x / exo_clusterDim,
            gridDim.x / exo_clusterDim,
            exo_cudaTasksLo_task, exo_cudaTasksHi_task,
            exo_deviceArgs);
        while (exo_taskGenerator.prepare_next_task()) {
          exo_deviceTask(exo_smem, exo_syncState, exo_deviceArgs, exo_taskGenerator.get_next_task(), exo_excutLog);
        }
      }
    }
  }
}
