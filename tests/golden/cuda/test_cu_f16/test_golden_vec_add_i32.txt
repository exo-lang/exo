// cu_f16_test_vec_add_i32(
//     N : size,
//     h_x : i32[N] @DRAM,
//     h_y : i32[N] @DRAM,
//     h_out : f32[N] @DRAM
// )
void cu_f16_test_vec_add_i32( void *ctxt, int_fast32_t N, const int32_t* h_x, const int32_t* h_y, float* h_out ) {
EXO_ASSUME(N % 256 == 0);
int32_t *d_x = (int32_t*) exo_cudaMallocAsync(N * sizeof(*d_x), exo_cudaStream);
int32_t *d_y = (int32_t*) exo_cudaMallocAsync(N * sizeof(*d_y), exo_cudaStream);
float *d_out = (float*) exo_cudaMallocAsync(N * sizeof(*d_out), exo_cudaStream);
cudaMemcpyAsync(&d_x[0], &h_x[0], 4 * N, cudaMemcpyHostToDevice, exo_cudaStream);
cudaMemcpyAsync(&d_y[0], &h_y[0], 4 * N, cudaMemcpyHostToDevice, exo_cudaStream);
{
  struct exo_CudaDeviceArgs0_cu_f16_test_vec_add_i32 exo_deviceArgs = {
    N, d_x, d_y, d_out, exo_excut_get_device_log()
  };
  exo_cudaLaunch0_cu_f16_test_vec_add_i32(exo_cudaStream, &exo_deviceArgs);
}
exo_cudaFreeAsync(d_y, exo_cudaStream);
exo_cudaFreeAsync(d_x, exo_cudaStream);
cudaMemcpyAsync(&h_out[0], &d_out[0], 4 * N, cudaMemcpyDeviceToHost, exo_cudaStream);
exo_cudaFreeAsync(d_out, exo_cudaStream);
}

inline namespace exo_CudaInline_test_golden_vec_add_i32 {
struct exo_Cuda0_cu_f16_test_vec_add_i32
{
  using exo_DeviceArgs = exo_CudaDeviceArgs0_cu_f16_test_vec_add_i32;

  static constexpr uint32_t exo_blockDim = 256;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 0;


  struct exo_Task
  {
    int_fast32_t task;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_task;
    uint32_t exo_cudaTasksNum_task;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_task, int_fast32_t _exo_cudaTasksHi_task,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_task = _exo_cudaTasksLo_task;
      exo_cudaTasksNum_task = static_cast<uint32_t>(_exo_cudaTasksHi_task - _exo_cudaTasksLo_task);
      exo_taskCount *= exo_cudaTasksNum_task;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.task = exo_cudaTasksLo_task + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_task);
      exo_tmp /= exo_cudaTasksNum_task;
      return exo_task;
    }
  };

  struct exo_SyncState
  {

  };

  static inline const char*& exo_FILE()
  {
    static const char* name = __FILE__;
    return name;
  }

  static void
  exo_cudaLaunch(cudaStream_t exo_cudaStream, const exo_DeviceArgs& exo_deviceArgs);

  static __device__ __forceinline__ void
  exo_deviceSetup(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceMainLoop(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

  static __device__ __forceinline__ void
  exo_deviceTask(
      char* exo_smem,
      exo_SyncState& exo_syncState,
      const exo_DeviceArgs& exo_deviceArgs,
      exo_Task exo_task,
      exo_ExcutThreadLog exo_excutLog={});
};

inline void
exo_CudaInline_test_golden_vec_add_i32::exo_Cuda0_cu_f16_test_vec_add_i32::exo_cudaLaunch(
    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_golden_vec_add_i32;
  cudaFuncSetAttribute(exo_deviceFunction0_cu_f16_test_vec_add_i32, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 1u;

  cudaLaunchConfig_t exo_launchConfig = {};
  exo_launchConfig.gridDim = dim3(exo_gridDim, 1, 1);
  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

  cudaLaunchKernelEx(&exo_launchConfig, exo_deviceFunction0_cu_f16_test_vec_add_i32, exo_deviceArgs);

  exo_excut_flush_device_log(
      exo_cudaStream, exo_gridDim, exo_blockDim,
      exo_CudaUtil::exo_excut_str_id_count, exo_CudaUtil::exo_excut_str_table,
      1, &exo_FILE());
}

__device__ __forceinline__ void
exo_CudaInline_test_golden_vec_add_i32::exo_Cuda0_cu_f16_test_vec_add_i32::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_golden_vec_add_i32;
  exo_CudaRmemPacked32_i32 rmem_x;
  // cuda_threads(0, 256, unit=cuda_thread)
  if ([[maybe_unused]] int exo_1thr_tid = threadIdx.x; 1) {
    memcpy(&rmem_x.ptx_data, &exo_deviceArgs.d_x[exo_1thr_tid + 256 * exo_task.task], 4);
    exo_CudaRmemPacked32_i32 rmem_y;
    memcpy(&rmem_y.ptx_data, &exo_deviceArgs.d_y[exo_1thr_tid + 256 * exo_task.task], 4);
    for (int pack_st_out = 0; pack_st_out < 1; pack_st_out++) {
      exo_deviceArgs.d_out[pack_st_out + exo_1thr_tid + 256 * exo_task.task] = (float)(rmem_x[pack_st_out] + rmem_y[pack_st_out]);
    }
  }
}
__device__ __forceinline__ void
exo_CudaInline_test_golden_vec_add_i32::exo_Cuda0_cu_f16_test_vec_add_i32::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_golden_vec_add_i32;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 256) {
      {
        int_fast32_t exo_cudaTasksLo_task = 0;
        int_fast32_t exo_cudaTasksHi_task = ((exo_deviceArgs.N) / (256));
        exo_TaskGenerator exo_taskGenerator(
            blockIdx.x / exo_clusterDim,
            gridDim.x / exo_clusterDim,
            exo_cudaTasksLo_task, exo_cudaTasksHi_task,
            exo_deviceArgs);
        while (exo_taskGenerator.prepare_next_task()) {
          exo_deviceTask(exo_smem, exo_syncState, exo_deviceArgs, exo_taskGenerator.get_next_task(), exo_excutLog);
        }
      }
    }
  }
}
