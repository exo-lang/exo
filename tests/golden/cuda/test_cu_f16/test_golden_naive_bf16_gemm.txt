#ifdef __CUDACC__
template <typename T>
struct exo_Sm90_SW128 {
    T data;

    static EXO_CUDA_INLINE exo_Sm90_SW128<T>* swizzle_pointer(uintptr_t addr)
    {
        // Adapted from ThunderKittens appendix which actually documents CUDA correctly.
        addr = (addr ^ (((addr % 1024) >> 7) << 4));
        return reinterpret_cast<exo_Sm90_SW128*>(addr);
    }

    static constexpr uint64_t get_swizzle_bits()
    {
        return 1;
    }

    EXO_CUDA_INLINE const T& swizzle_get() const
    {
        return swizzle_pointer(reinterpret_cast<uintptr_t>(&data))->data;
    }

    EXO_CUDA_INLINE T& swizzle_get()
    {
        return swizzle_pointer(reinterpret_cast<uintptr_t>(&data))->data;
    }
};

// f16_naive_gemm_bf16_bf16_f32(
//     M : size,
//     N : size,
//     K : size,
//     h_A : bf16[M, K] @DRAM,
//     h_B : bf16[N, K] @DRAM,
//     h_C : f32[N, M] @DRAM
// )
void f16_naive_gemm_bf16_bf16_f32( void *ctxt, int_fast32_t M, int_fast32_t N, int_fast32_t K, const exo_bf16* h_A, const exo_bf16* h_B, float* h_C ) {
EXO_ASSUME(M % 128 == 0);
EXO_ASSUME(N % 128 == 0);
EXO_ASSUME(K % 32 == 0);
exo_bf16 *d_A = (exo_bf16*) exo_cudaMallocAsync(M * K * sizeof(*d_A), exo_cudaStream);
exo_bf16 *d_B = (exo_bf16*) exo_cudaMallocAsync(N * K * sizeof(*d_B), exo_cudaStream);
float *d_C = (float*) exo_cudaMallocAsync(N * M * sizeof(*d_C), exo_cudaStream);
cudaMemcpyAsync(&d_A[0], &h_A[0], 2 * (M + 0) * (K + 0), cudaMemcpyHostToDevice, exo_cudaStream);
cudaMemcpyAsync(&d_B[0], &h_B[0], 2 * (N + 0) * (K + 0), cudaMemcpyHostToDevice, exo_cudaStream);
{
  struct exo_CudaDeviceArgs0_f16_naive_gemm_bf16_bf16_f32 exo_deviceArgs = {
    M, N, K, d_A, d_B, d_C, exo_excut_get_device_log()
  };
  exo_cudaLaunch0_f16_naive_gemm_bf16_bf16_f32(exo_cudaStream, &exo_deviceArgs);
}
exo_cudaFreeAsync(d_B, exo_cudaStream);
exo_cudaFreeAsync(d_A, exo_cudaStream);
cudaMemcpyAsync(&h_C[0], &d_C[0], 4 * N * M, cudaMemcpyDeviceToHost, exo_cudaStream);
exo_cudaFreeAsync(d_C, exo_cudaStream);
}

// We need this inline namespace to avoid ODR problems in pytest.
inline namespace exo_CudaInline_test_golden_naive_bf16_gemm {
struct exo_Cuda0_f16_naive_gemm_bf16_bf16_f32
{
  using exo_DeviceArgs = exo_CudaDeviceArgs0_f16_naive_gemm_bf16_bf16_f32;

  static constexpr uint32_t exo_blockDim = 128;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 16384;
  static constexpr unsigned exo_smemOffset0_s_A = 0;  // 8192-byte allocation
  static constexpr unsigned exo_smemOffset1_s_B = 8192;  // 8192-byte allocation

  struct exo_Task
  {
    int_fast32_t m_cta;
    int_fast32_t n_cta;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_m_cta;
    uint32_t exo_cudaTasksNum_m_cta;
    int_fast32_t exo_cudaTasksLo_n_cta;
    uint32_t exo_cudaTasksNum_n_cta;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_m_cta, int_fast32_t _exo_cudaTasksHi_m_cta,
        int_fast32_t _exo_cudaTasksLo_n_cta, int_fast32_t _exo_cudaTasksHi_n_cta,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_m_cta = _exo_cudaTasksLo_m_cta;
      exo_cudaTasksNum_m_cta = static_cast<uint32_t>(_exo_cudaTasksHi_m_cta - _exo_cudaTasksLo_m_cta);
      exo_taskCount *= exo_cudaTasksNum_m_cta;
      exo_cudaTasksLo_n_cta = _exo_cudaTasksLo_n_cta;
      exo_cudaTasksNum_n_cta = static_cast<uint32_t>(_exo_cudaTasksHi_n_cta - _exo_cudaTasksLo_n_cta);
      exo_taskCount *= exo_cudaTasksNum_n_cta;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.n_cta = exo_cudaTasksLo_n_cta + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_n_cta);
      exo_tmp /= exo_cudaTasksNum_n_cta;
      exo_task.m_cta = exo_cudaTasksLo_m_cta + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_m_cta);
      exo_tmp /= exo_cudaTasksNum_m_cta;
      return exo_task;
    }
  };

  struct exo_SyncState
  {

  };

  static inline const char*& exo_FILE()
  {
    static const char* name = __FILE__;
    return name;
  }

  static void
  exo_cudaLaunch(cudaStream_t exo_cudaStream, const exo_DeviceArgs& exo_deviceArgs);

  static __device__ __forceinline__ void
  exo_deviceSetup(char* exo_smem, const exo_DeviceArgs& exo_deviceArgs, exo_ExcutThreadLog exo_excutLog={});

inline void
exo_CudaInline_test_golden_naive_bf16_gemm::exo_Cuda0_f16_naive_gemm_bf16_bf16_f32::exo_cudaLaunch(
    cudaStream_t exo_cudaStream,
    const exo_DeviceArgs& exo_deviceArgs)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_golden_naive_bf16_gemm;
  cudaFuncSetAttribute(exo_deviceFunction0_f16_naive_gemm_bf16_bf16_f32, cudaFuncAttributeMaxDynamicSharedMemorySize, exo_smemBytes);
  // TODO how expensive is it to query this every time?
  int exo_cudaDevice;
  cudaGetDevice(&exo_cudaDevice);
  int exo_SMs;
  cudaDeviceGetAttribute(&exo_SMs, cudaDevAttrMultiProcessorCount, exo_cudaDevice);
  const unsigned exo_gridDim = (unsigned(exo_SMs) & ~(exo_clusterDim - 1)) * 3u;

  cudaLaunchConfig_t exo_launchConfig = {};
  exo_launchConfig.gridDim = dim3(exo_gridDim, 1, 1);
  exo_launchConfig.blockDim = dim3(exo_blockDim, 1, 1);
  exo_launchConfig.dynamicSmemBytes = exo_smemBytes;
  exo_launchConfig.stream = exo_cudaStream;

  cudaLaunchKernelEx(&exo_launchConfig, exo_deviceFunction0_f16_naive_gemm_bf16_bf16_f32, exo_deviceArgs);

  exo_excut_flush_device_log(
      exo_cudaStream, exo_gridDim, exo_blockDim,
      exo_CudaUtil::exo_excut_str_id_count, exo_CudaUtil::exo_excut_str_table,
      1, &exo_FILE());
}

  float r_C[128];
  // cuda_threads(0, 128, unit=cuda_thread)
  if ([[maybe_unused]] int exo_1thr_m = threadIdx.x; 1) {
    for (int n = 0; n < 128; n++) {

    }

        }
      }
    }

    asm volatile(
      "barrier.cta.sync 0;"
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
    exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
    // cuda_threads(0, 128, unit=cuda_thread)
    if ([[maybe_unused]] int exo_1thr_m = threadIdx.x; 1) {
      for (int n = 0; n < 128; n++) {
        for (int k = 0; k < 32; k++) {

          a_f32 = (float)(s_A[exo_1thr_m * 32 + k].swizzle_get());
          b_f32 = (float)(s_B[(n) * (32) + k]);
          r_C[n] += a_f32 * b_f32;
        }
      }
    }
    // Fence(cuda_in_order, cuda_in_order)
    asm volatile(
      "barrier.cta.sync 0;"
    );

  if ([[maybe_unused]] int exo_1thr_m = threadIdx.x; 1) {
    for (int n = 0; n < 128; n++) {
