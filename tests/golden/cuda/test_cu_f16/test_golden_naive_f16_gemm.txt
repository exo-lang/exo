#ifndef EXO_MEMORY_GLOBAL_Sm90_SmemSwizzled_128
#define EXO_MEMORY_GLOBAL_Sm90_SmemSwizzled_128

#ifdef __CUDACC__
template <typename T>
struct exo_Sm90_SW128 {
    T data;

    static EXO_CUDA_INLINE exo_Sm90_SW128<T>* swizzle_pointer(uintptr_t addr)
    {
        // Adapted from ThunderKittens appendix which actually documents CUDA correctly.
        uint32_t shr = uint32_t(addr) >> 3;
        const uint32_t mask = 112;
        addr = addr ^ (shr & mask);
        return reinterpret_cast<exo_Sm90_SW128*>(addr);
    }

    static __host__ __device__ constexpr uint64_t get_swizzle_bits()
    {
        return 1;
    }

    static __host__ __device__ constexpr int get_swizzle_bytes()
    {
        return 128;
    }

    EXO_CUDA_INLINE const T& swizzle_get() const
    {
        return swizzle_pointer(reinterpret_cast<uintptr_t>(&data))->data;
    }

    EXO_CUDA_INLINE T& swizzle_get()
    {
        return swizzle_pointer(reinterpret_cast<uintptr_t>(&data))->data;
    }
};

// Element type for Sm90_SmemSwizzled(128) allocations where the
// last 3 array extents are [TileOuterCols, TileRows, TileInnerCols]
// (left-padded with 1 for 0D, 1D, 2D allocations).
template <typename T, int TileOuterCols, int TileRows, int TileInnerCols>
struct exo_Sm90_SW128_tiled: public exo_Sm90_SW128<T>
{
    // Reinterpret-cast to kittens::st_subtile view of this shared memory.
    //
    // For better or worse, kittens 2D tile is expressed as a 3D tile in Exo-GPU.
    // `this` is assumed to point-to the NON-SWIZZLED base address of a complete
    // 3D tile. This returns a subtile view whose
    //
    // * size/extents are given by the Subtile template parameters
    //
    // * base offset from the `this` tile is given by runtime int offsets
    //
    // Assume for this discussion that the tile is row-major. Then,
    //
    // * Let swizzle_elements = 128 / sizeof(T)
    //
    // * The 3D Exo-GPU tile is of size [TileOuterCols, TileRows, TileInnerCols],
    //   where TileInnerCols == swizzle_elements. If not, then this tile
    //   is incompatible with ThunderKittens, TMA, and wgmma.
    //
    // * The value at coordinates (r, c) in the 2D [Rows, Cols] tile is stored at
    //   address swizzle_pointer(
    //          tile_base_addr
    //          + sizeof(T) * (c / TileInnerCols) * TileRows * TileInnerCols
    //          + sizeof(T) * (r) * TileInnerCols
    //          + sizeof(T) * (c % TileInnerCols))
    //   Basically, the column is simultaneously the fastest (%) and slowest (/) dimension
    //
    // TMA: If TileOuterCols != 1, then ThunderKittens is capable of doing a single TMA
    // copy to load/store the tile, but Exo-GPU may have to issue multiple.
    template <
        int SubtileOuterCols,
        int SubtileRows,
        int SubtileInnerCols,
        template <int, int, bool, int> class st_typed>
    EXO_CUDA_INLINE auto as_tk_subtile(int col_outer_offset, int row_offset, int col_inner_offset) const
    {
        static_assert(
            TileInnerCols * sizeof(T) == 128,
            "Exo-GPU instr didn't assert strides properly."
            " This is needed to match kittens swizzle automation"
        );
        static_assert(
            SubtileInnerCols == TileInnerCols || SubtileOuterCols == 1,
            "Exo-GPU instr used wrong SubtileInnerCols (window size last dimension)."
            " This is needed for Exo-GPU and kittens to agree on column tiling"
        );
        using st_t = st_typed<TileRows, TileOuterCols * TileInnerCols, true, 0>;
        st_t* p_tile = const_cast<st_t*>(reinterpret_cast<const st_t*>(this));
        // Note, subtile implicitly multiplies (r, c) by the subtile size.
        // We have to work around this!
        // Also, can't mention ::kittens here, because it may not be included.
        // Also also, explicit kittens swizzle is broken in this function.
        auto subtile = p_tile->template subtile<SubtileRows, SubtileOuterCols * SubtileInnerCols>(int2(0, 0));
        subtile.row_offset = static_cast<int>(row_offset);
        subtile.col_offset = static_cast<int>(col_outer_offset) * TileInnerCols + static_cast<int>(col_inner_offset);
        return subtile;
    }
};

template <typename PtxType, typename Scalar, typename PackedStruct>
struct exo_CudaRmemPacked32
{
    static_assert(sizeof(PtxType) == sizeof(PackedStruct));
    PtxType ptx_data;

    template <typename Index>
    __device__ auto operator[] (Index i) const -> Scalar
    {
        if constexpr (sizeof(Scalar) == 4)
            return ptx_data;
        else if (i == 0)
            return reinterpret_cast<const PackedStruct*>(&ptx_data)->x;
        else
            return reinterpret_cast<const PackedStruct*>(&ptx_data)->y;
    }
};

    exo_f16* d_B;  // d_B : f16[N, K] @CudaGmemLinear
    float* d_C;  // d_C : f32[N, M] @CudaGmemLinear
    EXO_EXCUT_DEVICE_LOG_MEMBER  // for Exo pytest (exo_excut.h)

// f16_naive_gemm_f16_f16_f32(
//     M : size,
//     N : size,
//     K : size,
//     h_A : f16[M, K] @DRAM,
//     h_B : f16[N, K] @DRAM,
//     h_C : f32[N, M] @DRAM
// )
void f16_naive_gemm_f16_f16_f32( void *ctxt, int_fast32_t M, int_fast32_t N, int_fast32_t K, const exo_f16* h_A, const exo_f16* h_B, float* h_C ) {
  EXO_ASSUME(M % 128 == 0);
  EXO_ASSUME(N % 128 == 0);
  EXO_ASSUME(K % 32 == 0);
  exo_f16 *d_A = (exo_f16*) exo_cudaMallocAsync(M * K * sizeof(*d_A), exo_cudaStream);
  exo_f16 *d_B = (exo_f16*) exo_cudaMallocAsync(N * K * sizeof(*d_B), exo_cudaStream);
  float *d_C = (float*) exo_cudaMallocAsync(N * M * sizeof(*d_C), exo_cudaStream);
  cudaMemcpyAsync(
    (&d_A[0]),
    (&h_A[0]),
    M * K * 2,
    cudaMemcpyHostToDevice,
    exo_cudaStream);
  cudaMemcpyAsync(
    (&d_B[0]),
    (&h_B[0]),
    N * K * 2,
    cudaMemcpyHostToDevice,
    exo_cudaStream);
  {
    struct exo_CudaDeviceArgs0_f16_naive_gemm_f16_f16_f32 exo_deviceArgs = {
      M, N, K, d_A, d_B, d_C, exo_excut_get_device_log()
    };
    exo_cudaLaunch0_f16_naive_gemm_f16_f16_f32(exo_cudaStream, &exo_deviceArgs);
  }
  exo_cudaFreeAsync(d_B, exo_cudaStream);
  exo_cudaFreeAsync(d_A, exo_cudaStream);
  cudaMemcpyAsync(
    (&h_C[0]),
    (&d_C[0]),
    N * M * 4,
    cudaMemcpyDeviceToHost,
    exo_cudaStream);
  exo_cudaFreeAsync(d_C, exo_cudaStream);
}

        st_t* p_tile = const_cast<st_t*>(reinterpret_cast<const st_t*>(this));
        // Note, subtile implicitly multiplies (r, c) by the subtile size.
        // We have to work around this!
        // Also, can't mention ::kittens here, because it may not be included.
        // Also also, explicit kittens swizzle is broken in this function.
        auto subtile = p_tile->template subtile<SubtileRows, SubtileOuterCols * SubtileInnerCols>(int2(0, 0));
        subtile.row_offset = static_cast<int>(row_offset);
        subtile.col_offset = static_cast<int>(col_outer_offset) * TileInnerCols + static_cast<int>(col_inner_offset);
        return subtile;
    }
};

  using exo_DeviceArgs = exo_CudaDeviceArgs0_f16_naive_gemm_f16_f16_f32;

  static constexpr uint32_t exo_blockDim = 128;
  static constexpr uint32_t exo_clusterDim = 1;

  static constexpr unsigned exo_smemBytes = 16384;
  static constexpr unsigned exo_smemOffset0_s_A = 0;  // 8192-byte allocation
  static constexpr unsigned exo_smemOffset1_s_B = 8192;  // 8192-byte allocation

  struct exo_Task
  {
    int_fast32_t m_cta;
    int_fast32_t n_cta;
  };

  struct exo_TaskGenerator
  {
    uint32_t exo_taskIndex;
    uint32_t exo_numClusters;
    uint32_t exo_taskCount;
    int_fast32_t exo_cudaTasksLo_m_cta;
    uint32_t exo_cudaTasksNum_m_cta;
    int_fast32_t exo_cudaTasksLo_n_cta;
    uint32_t exo_cudaTasksNum_n_cta;
    EXO_CUDA_INLINE exo_TaskGenerator(
        uint32_t cluster_index, uint32_t num_clusters,
        int_fast32_t _exo_cudaTasksLo_m_cta, int_fast32_t _exo_cudaTasksHi_m_cta,
        int_fast32_t _exo_cudaTasksLo_n_cta, int_fast32_t _exo_cudaTasksHi_n_cta,
        const exo_DeviceArgs&)
    {
      exo_taskIndex = cluster_index;
      exo_numClusters = num_clusters;
      exo_taskCount = 1;
      exo_cudaTasksLo_m_cta = _exo_cudaTasksLo_m_cta;
      exo_cudaTasksNum_m_cta = static_cast<uint32_t>(_exo_cudaTasksHi_m_cta - _exo_cudaTasksLo_m_cta);
      exo_taskCount *= exo_cudaTasksNum_m_cta;
      exo_cudaTasksLo_n_cta = _exo_cudaTasksLo_n_cta;
      exo_cudaTasksNum_n_cta = static_cast<uint32_t>(_exo_cudaTasksHi_n_cta - _exo_cudaTasksLo_n_cta);
      exo_taskCount *= exo_cudaTasksNum_n_cta;
    }
    [[nodiscard]] EXO_CUDA_INLINE bool prepare_next_task()
    {
      return exo_taskIndex < exo_taskCount;
    }
    EXO_CUDA_INLINE exo_Task get_next_task()
    {
      exo_Task exo_task;
      uint32_t exo_tmp = exo_taskIndex;
      exo_taskIndex += exo_numClusters;
      exo_task.n_cta = exo_cudaTasksLo_n_cta + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_n_cta);
      exo_tmp /= exo_cudaTasksNum_n_cta;
      exo_task.m_cta = exo_cudaTasksLo_m_cta + static_cast<int_fast32_t>(exo_tmp % exo_cudaTasksNum_m_cta);
      exo_tmp /= exo_cudaTasksNum_m_cta;
      return exo_task;
    }
  };

  struct exo_SyncState
  {

  };

    return name;

__device__ __forceinline__ void
exo_CudaInline_test_golden_naive_f16_gemm::exo_Cuda0_f16_naive_gemm_f16_f16_f32::exo_deviceTask(
    char* exo_smem,
    exo_SyncState& exo_syncState,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_Task exo_task,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_golden_naive_f16_gemm;
  auto& s_A = reinterpret_cast<exo_Sm90_SW128_tiled<exo_f16, 1, 128, 32> (&)[]>(exo_smem[exo_smemOffset0_s_A]);
  auto& s_B = reinterpret_cast<exo_f16 (&) [4096]>(exo_smem[exo_smemOffset1_s_B]);
  float r_C[128];
  // cuda_threads(0, 128, unit=cuda_thread)
  if ([[maybe_unused]] int exo_1thr_m = threadIdx.x; 1) {
    for (int n = 0; n < 128; n++) {
      r_C[n] = 0.0f;
    }
  }
  for (int ks = 0; ks < ((exo_deviceArgs.K) / (32)); ks++) {
    struct exo_win_2f16 d_A_tile = (struct exo_win_2f16) { &exo_deviceArgs.d_A[(128 * exo_task.m_cta) * exo_deviceArgs.K + (32 * ks)], {exo_deviceArgs.K, 1} };
    struct exo_win_2f16 d_B_tile = (struct exo_win_2f16) { &exo_deviceArgs.d_B[(128 * exo_task.n_cta) * exo_deviceArgs.K + (32 * ks)], {exo_deviceArgs.K, 1} };
    for (int ms = 0; ms < 16; ms++) {
      // cuda_threads(0, 8, unit=16 * cuda_thread)
      if ([[maybe_unused]] int exo_16thr_mt = (threadIdx.x / 16); 1) {
        // cuda_threads(0, 16, unit=cuda_thread)
        if ([[maybe_unused]] int exo_1thr_kt = (threadIdx.x % 16); 1) {
          exo_CudaRmemPacked32_f16 tmp_A;
          exo_CudaRmemPacked32_f16 tmp_B;
          tmp_A.ptx_data = *reinterpret_cast<const decltype(tmp_A.ptx_data)*>((&d_A_tile.data[(exo_16thr_mt + 8 * ms) * d_A_tile.strides[0] + (2 * exo_1thr_kt) * d_A_tile.strides[1]]));
          tmp_B.ptx_data = *reinterpret_cast<const decltype(tmp_B.ptx_data)*>((&d_B_tile.data[(exo_16thr_mt + 8 * ms) * d_B_tile.strides[0] + (2 * exo_1thr_kt) * d_B_tile.strides[1]]));
          *reinterpret_cast<decltype(tmp_A.ptx_data)*>((&s_A[((exo_16thr_mt + 8 * ms) * 32 + 2 * exo_1thr_kt)].swizzle_get())) = tmp_A.ptx_data;
          *reinterpret_cast<decltype(tmp_B.ptx_data)*>((&s_B[(exo_16thr_mt + 8 * ms) * 32 + (2 * exo_1thr_kt)])) = tmp_B.ptx_data;
        }
      }
    }
    // Fence(cuda_in_order, cuda_in_order)
    asm volatile(
      "barrier.cta.sync 0;"
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
    exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
    // cuda_threads(0, 128, unit=cuda_thread)
    if ([[maybe_unused]] int exo_1thr_m = threadIdx.x; 1) {
      for (int n = 0; n < 128; n++) {
        for (int k = 0; k < 32; k++) {
          float a_f32;
          float b_f32;
          a_f32 = (float)(s_A[(exo_1thr_m * 32 + k)].swizzle_get());
          b_f32 = (float)(s_B[n * 32 + k]);
          r_C[n] += a_f32 * b_f32;
        }
      }
    }
    // Fence(cuda_in_order, cuda_in_order)
    asm volatile(
      "barrier.cta.sync 0;"
    );
    exo_excutLog.log_action(EXO_EXCUT_STR_ID(barrier_cta_sync), 0, __LINE__);
    exo_excutLog.log_u32_arg(static_cast<uint32_t>(0));
  }
  // cuda_threads(0, 128, unit=cuda_thread)
  if ([[maybe_unused]] int exo_1thr_m = threadIdx.x; 1) {
    for (int n = 0; n < 128; n++) {
      exo_deviceArgs.d_C[(n + 128 * exo_task.n_cta) * exo_deviceArgs.M + (exo_1thr_m + 128 * exo_task.m_cta)] = r_C[n];
    }
  }
}
__device__ __forceinline__ void
exo_CudaInline_test_golden_naive_f16_gemm::exo_Cuda0_f16_naive_gemm_f16_f16_f32::exo_deviceMainLoop(
    char* exo_smem,
    const exo_DeviceArgs& exo_deviceArgs,
    exo_ExcutThreadLog exo_excutLog)
{
  namespace exo_CudaUtil = exo_CudaUtil_test_golden_naive_f16_gemm;
  exo_SyncState exo_syncState{};
  int32_t nreg;
  nreg = ((int32_t) 0);
  if (nreg == ((int32_t) 0)) {
    if (int tmp = threadIdx.x; tmp >= 0 && tmp < 128) {
      {
        int_fast32_t exo_cudaTasksLo_m_cta = 0;
        int_fast32_t exo_cudaTasksHi_m_cta = ((exo_deviceArgs.M) / (128));
        {
          int_fast32_t exo_cudaTasksLo_n_cta = 0;
          int_fast32_t exo_cudaTasksHi_n_cta = ((exo_deviceArgs.N) / (128));
          exo_TaskGenerator exo_taskGenerator(
              blockIdx.x / exo_clusterDim,
              gridDim.x / exo_clusterDim,
              exo_cudaTasksLo_m_cta, exo_cudaTasksHi_m_cta,
              exo_cudaTasksLo_n_cta, exo_cudaTasksHi_n_cta,
              exo_deviceArgs);
          while (exo_taskGenerator.prepare_next_task()) {
            exo_deviceTask(exo_smem, exo_syncState, exo_deviceArgs, exo_taskGenerator.get_next_task(), exo_excutLog);
          }
        }
      }
    }
  }
}
