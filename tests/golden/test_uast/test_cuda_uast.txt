def tma_tester_gpu_proc(
        sum_tensorMap_window: [f32][744, 128] @ Sm90_tensorMap(128, 248, 32),
        d_x: [f32][744, 128] @ CudaGmemLinear,
        d_y: f32[744, 128] @ CudaGmemLinear):
    assert stride(d_x, 1) == 1
    assert stride(d_y, 1) == 1
    assert stride(sum_tensorMap_window, 1) == 1
    x_tensorMap_debug = d_x[:, :] @ Sm90_tensorMap(128, 248, 32)
    with CudaDeviceFunction(blockDim=256):
        for task_m in cuda_tasks(0, 3):
            for task_k in cuda_tasks(0, 4):
                smem_x: f32[248, 32] @ Sm90_SmemSwizzled(128, )
                smem_y: f32[248, 32] @ Sm90_SmemSwizzled(128, )
                smem_sum: f32[248, 32] @ Sm90_SmemSwizzled(128, )
                raw: barrier @ CudaMbarrier
                war: barrier(raw) @ CudaMbarrier
                with CudaWarps(0, 1):
                    Await(war, cuda_temporal, ~1)
                    x_input = x_tensorMap_debug[task_m * 248:task_m * 248 +
                                                248,
                                                task_k * 32:task_k * 32 + 32]
                    if 128 == 0:
                        Sm90_copy_tensor_to_smem_linear_2f32(
                            smem_x[:, :],
                            x_input,
                            size0=248,
                            size1=32,
                            smem_box=None) >> raw
                    else:
                        Sm90_copy_tensor_to_smem_swizzled_2f32(
                            smem_x[:, :],
                            x_input,
                            size0=248,
                            size1=32,
                            smem_box=None) >> raw
                    Arrive(cuda_temporal, 1) >> raw
                for m in cuda_threads(0, 248, unit=cuda_thread):
                    for ko in seq(0, 32 / 4):
                        Sm80_cp_async_f32(smem_y[m, ko * 4:ko * 4 + 4],
                                          d_y[task_m * 248 + m, task_k * 32 +
                                              ko * 4:task_k * 32 + ko * 4 + 4],
                                          size=4)
                Fence(Sm80_cp_async, cuda_in_order)
                Await(raw, cuda_in_order, ~0)
                for m in cuda_threads(0, 248, unit=cuda_thread):
                    for k in seq(0, 32):
                        smem_sum[m, k] = smem_x[m, k] + smem_y[m, k]
                Arrive(cuda_in_order, 1) >> war
                Fence(cuda_in_order, cuda_generic_and_async_proxy)
                with CudaWarps(0, 1):
                    tma_window = sum_tensorMap_window[task_m *
                                                      248:task_m * 248 + 248,
                                                      task_k * 32:task_k * 32 +
                                                      32]
                    if 128 == 0:
                        Sm90_copy_tensor_to_gmem_linear_2f32(tma_window[:, :],
                                                             smem_sum[:, :],
                                                             size0=248,
                                                             size1=32,
                                                             smem_box=None)
                    else:
                        Sm90_copy_tensor_to_gmem_swizzled_2f32(
                            tma_window[:, :],
                            smem_sum[:, :],
                            size0=248,
                            size1=32,
                            smem_box=None)
                    cg: barrier @ CudaCommitGroup
                    Arrive(tma_to_gmem_async, 1) >> cg
                    Await(cg, cuda_in_order, 0)
                Fence(cuda_in_order, cuda_in_order)
