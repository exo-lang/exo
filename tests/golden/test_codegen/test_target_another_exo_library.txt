#elif EXO_HAS_BUILTIN(__builtin_unreachable)
#  define EXO_ASSUME(expr) \
      ((void)((expr) ? 1 : (__builtin_unreachable(), 1)))
#else
#  define EXO_ASSUME(expr) ((void)(expr))
#endif

#include "foo.h"

// foo(
//     n : size,
//     x : f32[n] @DRAM
// )
void foo( void *ctxt, exo_control_t n, float* x ) {
  for (exo_control_t io = 0; io < ((n) / (4)); io++) {
    for (exo_control_t ii = 0; ii < 4; ii++) {
      x[(4 * io + ii)] = 1.0f;
    }
  }
  for (exo_control_t ii = 0; ii < n % 4; ii++) {
    x[(ii + (n >> 2) * 4)] = 1.0f;
  }
}
