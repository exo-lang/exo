// gemv(
//     M : size,
//     N : size,
//     A : f32[M, N] @DRAM,
//     x : f32[N] @DRAM,
//     y : f32[M] @DRAM
// )
void gemv( void *ctxt, exo_control_t M, exo_control_t N, const float* A, const float* x, float* y ) {
  EXO_ASSUME(M % 8 == 0);
  EXO_ASSUME(N % 8 == 0);
  for (exo_control_t io = 0; io < ((M) / (8)); io++) {
    for (exo_control_t jo = 0; jo < ((N) / (8)); jo++) {
      for (exo_control_t ii = 0; ii < 8; ii++) {
        for (exo_control_t ji = 0; ji < 8; ji++) {
          y[(8 * io + ii)] += A[(8 * io + ii) * N + (8 * jo + ji)] * x[(8 * jo + ji)];
        }
      }
    }
  }
}
