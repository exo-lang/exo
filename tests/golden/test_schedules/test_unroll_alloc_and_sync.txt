def bar(x: f32[256] @ CudaGmemLinear):
    with CudaDeviceFunction(blockDim=128):
        for task in cuda_tasks(0, 1):
            A: f32[128] @ CudaSmemLinear
            bar: barrier[1 @ (ring_buffer_by(depth=1))] @ CudaMbarrier
            for t in cuda_threads(0, 128, unit=cuda_thread):
                A[t] = x[0 * 128 + t]
            Arrive(cuda_in_order, 1) >> bar[0]
            Await(bar[0], cuda_in_order, 0)
            for t in cuda_threads(0, 128, unit=cuda_thread):
                x[0 * 128 + t] = A[127 - t]
            Fence(cuda_in_order, cuda_in_order)
            A_1: f32[128] @ CudaSmemLinear
            bar_1: barrier[1 @ (ring_buffer_by(depth=1))] @ CudaMbarrier
            for t in cuda_threads(0, 128, unit=cuda_thread):
                A_1[t] = x[1 * 128 + t]
            Arrive(cuda_in_order, 1) >> bar_1[0]
            Await(bar_1[0], cuda_in_order, 0)
            for t in cuda_threads(0, 128, unit=cuda_thread):
                x[1 * 128 + t] = A_1[127 - t]
            Fence(cuda_in_order, cuda_in_order)
